def get_app_preference(pref_id, default_value=None):
    try:
        pref_val = ExtAPI.Application.GetPreference(pref_id)
        if pref_val is None:
            return default_value
        pref_str = str(pref_val)
        if pref_str == "":
            return default_value
        return pref_str
    except Exception:
        pass

    # Fallback through Mechanical's jscript bridge.
    try:
        cmd = "returnFromScript(DS.Script.wb.PreferenceMgr.Preference('{0}'));".format(pref_id)
        pref_val = ExtAPI.Application.ScriptByName("jscript").ExecuteCommand(cmd)
        if pref_val is None:
            return default_value
        pref_str = str(pref_val)
        if pref_str == "":
            return default_value
        return pref_str
    except Exception:
        return default_value


def set_app_preference(pref_id, pref_value):
    value_str = str(pref_value)

    # First try direct API, if available.
    try:
        ExtAPI.Application.SetPreference(pref_id, value_str)
        return True
    except Exception:
        pass

    # Fallback through jscript/PreferenceMgr (used internally by Mechanical UI scripts).
    try:
        cmd = (
            "DS.Script.wb.PreferenceMgr.Preference('{0}') = '{1}';"
            "DS.Script.wb.PreferenceMgr.Save();"
            "returnFromScript(DS.Script.wb.PreferenceMgr.Preference('{0}'));"
        ).format(pref_id, value_str)
        new_val = ExtAPI.Application.ScriptByName("jscript").ExecuteCommand(cmd)
        if new_val is None:
            return True
        return str(new_val) == value_str
    except Exception:
        return False


def enter_no_graphics_preference_mode(progress_dialog=None):
    original_val = get_app_preference(PREF_LEVEL_OF_DETAIL, None)
    changed_ok = set_app_preference(PREF_LEVEL_OF_DETAIL, PREF_LEVEL_OF_DETAIL_NO_GRAPHICS)

    if changed_ok:
        log_message(
            "Set preference {0}=0 (No Graphics) for extraction.".format(PREF_LEVEL_OF_DETAIL),
            progress_dialog,
        )
    else:
        log_message(
            "Could not set {0}=0 via API in this session; using suspend-based redraw suppression only.".format(
                PREF_LEVEL_OF_DETAIL
            ),
            progress_dialog,
        )

    return {
        "pref_id": PREF_LEVEL_OF_DETAIL,
        "original_value": original_val,
        "restore_needed": (original_val is not None and str(original_val) != PREF_LEVEL_OF_DETAIL_NO_GRAPHICS),
        "set_succeeded": changed_ok,
    }


def restore_no_graphics_preference_mode(pref_state, progress_dialog=None):
    if pref_state is None:
        return
    if not pref_state.get("restore_needed", False):
        return

    pref_id = pref_state.get("pref_id", PREF_LEVEL_OF_DETAIL)
    original_val = pref_state.get("original_value", None)
    if original_val is None:
        return

    restored_ok = set_app_preference(pref_id, str(original_val))
    if restored_ok:
        log_message(
            "Restored preference {0} back to {1}.".format(pref_id, str(original_val)),
            progress_dialog,
        )
    else:
        log_message(
            "Warning: could not restore {0} to {1}. Please check Mechanical Options > Graphics.".format(
                pref_id, str(original_val)
            ),
            progress_dialog,
        )


def log_message(msg, progress_dialog=None):
    stamp = time.strftime("%H:%M:%S")
    line = "[{0}] {1}".format(stamp, msg)
    print(line)
    try:
        ExtAPI.Log.WriteMessage(line)
    except Exception:
        pass
    if progress_dialog is not None:
        try:
            progress_dialog.update_progress(msg)
        except Exception:
            pass

def format_available_ram_text(available_ram_bytes):
    if available_ram_bytes > 0:
        return "{0:.2f} GiB".format(bytes_to_gib(available_ram_bytes))
    return "unknown"


def get_available_ram_bytes():
    try:
        from Microsoft.VisualBasic.Devices import ComputerInfo
        return int(ComputerInfo().AvailablePhysicalMemory)
    except Exception:
        pass

    try:
        clr.AddReference("System")
        from System.Diagnostics import PerformanceCounter
        pc = PerformanceCounter("Memory", "Available MBytes")
        mb = float(pc.NextValue())
        return int(mb * 1024 * 1024)
    except Exception:
        pass

    try:
        clr.AddReference("System.Management")
        from System.Management import ManagementObjectSearcher
        searcher = ManagementObjectSearcher("SELECT FreePhysicalMemory FROM Win32_OperatingSystem")
        for mo in searcher.Get():
            free_kb = int(mo["FreePhysicalMemory"])
            return int(free_kb * 1024)
    except Exception:
        pass

    return -1


def force_memory_cleanup(wait_for_finalizers=False):
    try:
        gc.collect()
    except Exception:
        pass

    # Mechanical objects and arrays are .NET-backed; collect both runtimes.
    try:
        clr.AddReference("System")
        from System import GC as DotNetGC
        DotNetGC.Collect()
        if wait_for_finalizers:
            DotNetGC.WaitForPendingFinalizers()
            DotNetGC.Collect()
    except Exception:
        pass


def apply_low_ram_backpressure(progress_dialog=None, context_text=""):
    available_bytes = get_available_ram_bytes()
    if available_bytes > 0 and available_bytes < LOW_RAM_BACKPRESSURE_BYTES:
        msg = "Low available RAM detected ({0}). Running aggressive cleanup.".format(
            format_available_ram_text(available_bytes)
        )
        if context_text is not None and str(context_text).strip() != "":
            msg += " Context: {0}".format(str(context_text))
        log_message(msg, progress_dialog)
        force_memory_cleanup(True)
        time.sleep(0.2)
    return available_bytes


def estimate_mode_batch_size(node_count, component_count, remaining_modes):
    if remaining_modes <= 0:
        return 1, -1

    available_bytes = get_available_ram_bytes()
    if node_count <= 0 or component_count <= 0:
        return 1, available_bytes

    # Conservative estimate for Python + .NET object overhead.
    overhead_multiplier = 14.0
    bytes_per_mode = int(node_count * component_count * 8.0 * overhead_multiplier)

    if bytes_per_mode >= VERY_LARGE_MODE_BYTES_THRESHOLD:
        mode_size_cap = 1
    elif bytes_per_mode >= LARGE_MODE_BYTES_THRESHOLD:
        mode_size_cap = 2
    else:
        mode_size_cap = RAM_MAX_MODES_PER_BATCH

    if available_bytes > 0:
        safe_bytes = int(max(RAM_MIN_SAFE_BYTES, available_bytes * RAM_SAFETY_FRACTION))
        estimate = int(float(safe_bytes) / float(max(bytes_per_mode, 1)))
    else:
        estimate = 1

    if estimate < 1:
        estimate = 1
    if available_bytes > 0 and available_bytes < LOW_RAM_BACKPRESSURE_BYTES:
        mode_size_cap = min(mode_size_cap, 1)
    if estimate > mode_size_cap:
        estimate = mode_size_cap
    if estimate > remaining_modes:
        estimate = remaining_modes

    return estimate, available_bytes


def estimate_merge_group_limit(pending_count):
    if pending_count <= 0:
        return 1, -1

    available_bytes = get_available_ram_bytes()
    if available_bytes <= 0:
        estimate = MERGE_GROUP_LIMIT_UNKNOWN_RAM
    elif available_bytes < LOW_RAM_BACKPRESSURE_BYTES:
        estimate = MERGE_GROUP_LIMIT_LOW_RAM
    elif available_bytes < MERGE_GROUP_MID_RAM_THRESHOLD_BYTES:
        estimate = MERGE_GROUP_LIMIT_MID_RAM
    elif available_bytes < MERGE_GROUP_HIGH_RAM_THRESHOLD_BYTES:
        estimate = MERGE_GROUP_LIMIT_HIGH_RAM
    elif available_bytes < MERGE_GROUP_VERY_HIGH_RAM_THRESHOLD_BYTES:
        estimate = MERGE_GROUP_LIMIT_VERY_HIGH_RAM
    else:
        estimate = MERGE_GROUP_LIMIT_MAX

    if estimate < MERGE_GROUP_LIMIT_MIN:
        estimate = MERGE_GROUP_LIMIT_MIN
    if estimate > MERGE_GROUP_LIMIT_MAX:
        estimate = MERGE_GROUP_LIMIT_MAX
    if estimate > pending_count:
        estimate = pending_count

    return estimate, available_bytes


def is_modal_analysis_candidate(analysis):
    if analysis is None:
        return False

    analysis_type = getattr(analysis, "AnalysisType", None)

    if analysis_type is not None:
        # Enum numeric check (Modal=5 in Mechanical AnalysisType enum).
        int_candidates = [
            analysis_type,
            getattr(analysis_type, "Value", None),
            getattr(analysis_type, "value__", None),
        ]
        for candidate in int_candidates:
            try:
                if int(candidate) == ANALYSIS_TYPE_MODAL_VALUE:
                    return True
            except Exception:
                pass

        # Enum name check across Mechanical versions.
        at_text = str(analysis_type).strip().lower()
        if at_text == "modal" or at_text.endswith(".modal"):
            return True

    # Secondary filter by runtime .NET type name.
    type_text = str(analysis.GetType().FullName).lower()
    if "modal" in type_text:
        return True

    return False


def get_modal_analysis_candidates():
    analyses = as_list(ExtAPI.DataModel.Project.Model.Analyses)
    if len(analyses) == 0:
        raise Exception("No analysis found in Mechanical tree.")

    modal_analyses = []
    for analysis in analyses:
        if is_modal_analysis_candidate(analysis):
            modal_analyses.append(analysis)

    if len(modal_analyses) == 0:
        raise Exception("No modal analysis environment was detected in tree.")
    return modal_analyses


def get_analysis_solution_and_modes(selected_analysis):
    if selected_analysis is None:
        raise Exception("No analysis selected.")

    solution = selected_analysis.Solution
    if solution is None:
        raise Exception("Selected analysis has no Solution object.")

    results_data = selected_analysis.GetResultsData()
    if results_data is None:
        raise Exception("Results data unavailable. Solve or relink result file first.")

    try:
        mode_count = len(results_data.ListTimeFreq)
    finally:
        try:
            results_data.Dispose()
        except Exception:
            pass

    if mode_count <= 0:
        raise Exception("No modal sets were found in analysis '{0}'.".format(selected_analysis.Name))

    mode_ids = range(1, mode_count + 1)
    return solution, mode_ids

def get_global_coordinate_system_object():
    cs_children = as_list(ExtAPI.DataModel.Project.Model.CoordinateSystems.Children)
    if len(cs_children) > 0:
        for cs in cs_children:
            if bool(getattr(cs, "IsGlobal", False)):
                return cs
        return cs_children[0]

    all_cs = as_list(ExtAPI.DataModel.GetObjectsByType(DataModelObjectCategory.CoordinateSystem))
    if len(all_cs) > 0:
        for cs in all_cs:
            if bool(getattr(cs, "IsGlobal", False)):
                return cs
        return all_cs[0]

    return None


def is_nodal_named_selection(named_selection, mesh_data):
    try:
        location = named_selection.Location
        sel_type_text = str(location.SelectionType).lower().replace("_", "")
        if "meshnode" in sel_type_text:
            return True
        if "node" in sel_type_text and "mesh" in sel_type_text:
            return True

        ids = location.Ids
        if ids is None or len(ids) == 0:
            return False
        if mesh_data is None:
            return True

        probe_count = min(5, len(ids))
        hits = 0
        for i in xrange(probe_count):
            nid = int(ids[i])
            node = mesh_data.NodeById(nid)
            if node is not None:
                hits += 1
        return hits > 0
    except Exception:
        return False


def get_nodal_named_selections():
    all_ns = as_list(ExtAPI.DataModel.GetObjectsByType(DataModelObjectCategory.NamedSelection))
    all_ns_sorted = sorted(all_ns, key=lambda n: str(n.Name).lower())
    mesh_data = get_mesh_data_global()

    nodal = []
    for ns in all_ns_sorted:
        if is_nodal_named_selection(ns, mesh_data):
            nodal.append(ns)

    if len(nodal) == 0:
        # Fallback if nodal detection cannot confidently classify.
        return all_ns_sorted
    return nodal

    def _make_job_record(self):
        selected_analysis = self.selected_analysis()
        selected_ns = self.selected_named_selection()
        selected_kinds = self.selected_result_kinds()
        selected_base_output_dir = self.selected_base_output_dir()

        if selected_analysis is None:
            self._show_warning("Please select a modal analysis environment.", "Missing Analysis")
            return None
        if selected_ns is None:
            self._show_warning("Please select a nodal named selection.", "Missing Selection")
            return None
        if len(selected_kinds) == 0:
            self._show_warning("Select at least one result type to extract.", "Missing Result Type")
            return None

        try:
            selected_base_output_dir = self._ensure_output_base_dir(selected_base_output_dir)
        except Exception as out_ex:
            self._show_warning(
                "Could not access selected output folder: {0}".format(str(out_ex)),
                "Invalid Output Folder",
            )
            return None

        submitted_epoch = time.time()
        record = {
            "job_id": int(self._next_job_id),
            "analysis": selected_analysis,
            "analysis_name": str(selected_analysis.Name),
            "named_selection": selected_ns,
            "named_selection_name": str(selected_ns.Name),
            "result_kinds": list(selected_kinds),
            "use_no_graphics": self.use_no_graphics_enabled(),
            "base_output_dir": selected_base_output_dir,
            "submitted_epoch": submitted_epoch,
            "status": JOB_STATUS_PENDING,
            "output_dir": "",
            "output_csv_paths": [],
            "error": "",
            "started_epoch": 0.0,
            "completed_epoch": 0.0,
        }
        self._next_job_id += 1
        return record

    def _on_submit_job_clicked(self, sender, args):

            return
        new_job = self._make_job_record()
        if new_job is None:
            return

        self._queued_jobs.append(new_job)
        self._append_job_to_table(new_job)
        self._refresh_action_buttons()

        result_labels = get_result_kind_labels(new_job.get("result_kinds", []))
        base_output_dir = str(new_job.get("base_output_dir", "")).strip()

            base_output_dir = "(analysis WorkingDir)"

        queue_msg = "Job {0} submitted | Analysis: {1} | NS: {2} | Result types: {3} | No Graphics: {4} | Save base: {5}".format(

            new_job.get("analysis_name", ""),

            "; ".join(result_labels),
            format_yes_no(new_job.get("use_no_graphics", False)),

        )
        self.update_progress(queue_msg, 0, 1000)
        log_message(queue_msg, None)

    def _on_run_jobs_clicked(self, sender, args):
        if self._is_running:
            return
        if self._pending_jobs_count() == 0:
            self._show_warning("No pending jobs found. Use 'Submit Jobs' before running.", "No Pending Jobs")
            return

        self._is_running = True
        self._set_extract_controls_enabled(False)
        self.TopMost = True
        self._ensure_visible_foreground()
        start_msg = "Starting extraction jobs ({0} pending)...".format(self._pending_jobs_count())
        self.update_progress(start_msg, 0, 1000)

        try:
            self._run_callback(self)
        except Exception as ex:
            err = "Extraction jobs failed: {0}".format(str(ex))
            self.update_progress(err)
            self._show_error(err, "Modal Extractor Error")
        finally:
            self._is_running = False
            self.TopMost = False
            self._set_extract_controls_enabled(True)
            Application.DoEvents()

def apply_named_selection_scope(result_object, named_selection):
    result_object.ScopingMethod = GeometryDefineByType.Component
    try:
        result_object.Location = named_selection
        return
    except Exception:
        pass

def create_component_objects(solution, named_selection, result_kind):
    component_names = []
    component_objects = []

    with begin_transaction(True):
        if result_kind == RESULT_KIND_STRESS:
            definitions = [
                ("sx", "normal", NormalOrientationType.XAxis),
                ("sy", "normal", NormalOrientationType.YAxis),
                ("sz", "normal", NormalOrientationType.ZAxis),
                ("sxy", "shear", ShearOrientationType.XYPlane),
                ("syz", "shear", ShearOrientationType.YZPlane),
                ("sxz", "shear", ShearOrientationType.XZPlane),
            ]
            for comp_name, kind, orient in definitions:
                if kind == "normal":
                    obj = solution.AddNormalStress()
                    obj.NormalOrientation = orient
                else:
                    obj = solution.AddShearStress()
                    obj.ShearOrientation = orient
                obj.Name = "__TMP_MODAL_{0}".format(comp_name.upper())
                apply_named_selection_scope(obj, named_selection)
                component_names.append(comp_name)
                component_objects.append(obj)

        elif result_kind == RESULT_KIND_DEFORMATION:
            definitions = [
                ("ux", NormalOrientationType.XAxis),
                ("uy", NormalOrientationType.YAxis),
                ("uz", NormalOrientationType.ZAxis),
            ]
            for comp_name, orient in definitions:
                obj = solution.AddDirectionalDeformation()
                obj.NormalOrientation = orient
                obj.Name = "__TMP_MODAL_{0}".format(comp_name.upper())
                apply_named_selection_scope(obj, named_selection)
                component_names.append(comp_name)
                component_objects.append(obj)

        elif result_kind == RESULT_KIND_FORCE_MOMENT:
            global_cs = get_global_coordinate_system_object()
            if global_cs is None:
                raise Exception(
                    "Global Coordinate System was not found. Cannot extract ENFO/ENMO in global CS."
                )

            definitions = [
                ("enfox", "ENFOX"),
                ("enfoy", "ENFOY"),
                ("enfoz", "ENFOZ"),
                ("enmox", "ENMOX"),
                ("enmoy", "ENMOY"),
                ("enmoz", "ENMOZ"),
            ]
            for comp_name, expression in definitions:
                obj = solution.AddUserDefinedResult()
                obj.Expression = expression
                obj.CoordinateSystem = global_cs
                obj.Name = "__TMP_MODAL_{0}".format(comp_name.upper())
                apply_named_selection_scope(obj, named_selection)
                component_names.append(comp_name)
                component_objects.append(obj)
        else:
            raise Exception("Unknown result kind: {0}".format(result_kind))

    return component_names, component_objects


def cleanup_component_objects(component_objects):
    if component_objects is None:
        return
    try:
        graphics_ctx = begin_graphics_suspend()
    except Exception:
        graphics_ctx = _NoOpTransaction()
    try:
        tx_ctx = begin_transaction(True)
    except Exception:
        tx_ctx = _NoOpTransaction()

    with graphics_ctx:
        with tx_ctx:
            for obj in component_objects:
                try:
                    obj.Delete()
                except Exception:
                    pass
    force_memory_cleanup(True)

def get_scale_factors(result_kind, component_count, unit_text):
    factors = [1.0] * component_count
    unit_normalized = str(unit_text).strip().lower()

    # Keep same conversions as previous script behavior.
    if result_kind == RESULT_KIND_STRESS:
        if unit_normalized == "pa":
            factors = [1e-6] * component_count  # Pa -> MPa
    elif result_kind == RESULT_KIND_DEFORMATION:
        if unit_normalized == "m":
            factors = [1e3] * component_count  # m -> mm
    return factors


def extract_node_ids_from_result(result_object):
    independent = result_object.PlotData.Independents
    candidate_keys = ["Node", "Nodes", "NodeID", "Node ID"]
    available_keys = [str(k) for k in independent.Keys]

    for key in candidate_keys:
        if key in available_keys:
            arr = independent[key]
            return [int(x) for x in arr]

    if len(available_keys) > 0:
        arr = independent[available_keys[0]]
        return [int(x) for x in arr]

    raise Exception("Could not extract node IDs from PlotData.Independents.")


def extract_scaled_values(result_object, scale_factor):
    values_raw = result_object.PlotData.Dependents["Values"]
    return [float(v.Value if hasattr(v, "Value") else v) * scale_factor for v in values_raw]


def extract_aligned_values(result_object, base_node_ids, scale_factor, return_order_match=False):
    values = extract_scaled_values(result_object, scale_factor)
    current_node_ids = extract_node_ids_from_result(result_object)

    if len(current_node_ids) == len(base_node_ids):
        is_same_order = True
        if len(current_node_ids) > 0:
            if current_node_ids[0] != base_node_ids[0] or current_node_ids[-1] != base_node_ids[-1]:
                is_same_order = False
        if is_same_order:
            if return_order_match:
                return values, True
            return values

    value_map = {}
    n = min(len(current_node_ids), len(values))
    for i in xrange(n):
        value_map[int(current_node_ids[i])] = values[i]
    aligned_values = [value_map.get(int(nid), 0.0) for nid in base_node_ids]

    if return_order_match:
        return aligned_values, False
    return aligned_values


def get_node_coordinates(node_ids, progress_dialog=None):
    mesh_data = get_mesh_data_global()
    coords = []

    if mesh_data is None:
        log_message("MeshDataByName('Global') unavailable. Coordinate columns will be blank.", progress_dialog)
        for _ in node_ids:
            coords.append(("", "", ""))
        return coords

    total = len(node_ids)
    for i in xrange(total):
        nid = int(node_ids[i])
        node = mesh_data.NodeById(nid)
        if node is None:
            x = ""
            y = ""
            z = ""
        else:
            x = float(node.X.Value if hasattr(node.X, "Value") else node.X)
            y = float(node.Y.Value if hasattr(node.Y, "Value") else node.Y)
            z = float(node.Z.Value if hasattr(node.Z, "Value") else node.Z)
        coords.append((x, y, z))

        if i > 0 and i % 20000 == 0:
            msg = "Coordinates: {0}/{1} nodes".format(i, total)
            log_message(msg, progress_dialog)
            if progress_dialog is not None:
                progress_dialog.update_progress(msg)

    return coords

def write_batch_csv(path, batch_modes, component_names, node_ids, batch_data):
    with open(path, "wb") as f:
        writer = csv.writer(f)

        header = ["NodeID"]
        for mode in batch_modes:
            for comp_name in component_names:
                header.append("{0}_Mode{1}".format(comp_name, mode))
        writer.writerow(header)

        node_count = len(node_ids)
        comp_count = len(component_names)
        for row_index in xrange(node_count):
            row = [node_ids[row_index]]
            for mode in batch_modes:
                mode_components = batch_data[mode]
                for comp_idx in xrange(comp_count):
                    row.append(mode_components[comp_idx][row_index])
            writer.writerow(row)

def should_use_solution_evaluate(solution, temp_component_count):
    children_count = len(solution.Children)
    # If there are many existing results, evaluating the full solution can be slower.
    return children_count <= (temp_component_count + 4)


def merge_pass(base_csv_path, batch_csv_paths, output_csv_path, node_count, pass_index, total_passes, progress_dialog):
    base_handle = None
    out_handle = None
    batch_handles = []
    batch_readers = []
    warned_node_mismatch = False

    try:
        base_handle = open(base_csv_path, "rb")
        out_handle = open(output_csv_path, "wb")
        base_reader = csv.reader(base_handle)
        writer = csv.writer(out_handle)

        base_header = base_reader.next()
        merged_header = list(base_header)

        for batch_path in batch_csv_paths:
            h = open(batch_path, "rb")
            batch_handles.append(h)
            r = csv.reader(h)
            batch_readers.append(r)
            h_row = r.next()
            if len(h_row) > 1:
                merged_header.extend(h_row[1:])

        writer.writerow(merged_header)

        progress_interval = max(1, int(node_count / 20))
        for i in xrange(node_count):
            base_row = base_reader.next()
            base_node = base_row[0] if len(base_row) > 0 else ""
            out_row = list(base_row)

            for r in batch_readers:
                mode_row = r.next()
                if len(mode_row) > 0 and mode_row[0] != base_node and not warned_node_mismatch:
                    warned_node_mismatch = True
                    log_message(
                        "Warning: Node ID ordering mismatch detected while merging temporary files.",
                        progress_dialog,
                    )
                if len(mode_row) > 1:
                    out_row.extend(mode_row[1:])
            writer.writerow(out_row)

            if i > 0 and i % progress_interval == 0:
                pct = int((float(i) / float(node_count)) * 100.0)
                msg = "Merging pass {0}/{1}: {2}%".format(pass_index, total_passes, pct)
                if progress_dialog is not None:
                    progress_dialog.update_progress(msg)

    finally:
        if base_handle is not None:
            base_handle.close()
        if out_handle is not None:
            out_handle.close()
        for h in batch_handles:
            try:
                h.close()
            except Exception:
                pass


def safe_delete_file(path):
    try:
        if os.path.exists(path):
            os.remove(path)
    except Exception:
        pass


def cleanup_temp_folder(temp_dir):
    if temp_dir is None:
        return
    try:
        if os.path.isdir(temp_dir):
            for name in os.listdir(temp_dir):
                safe_delete_file(os.path.join(temp_dir, name))
            os.rmdir(temp_dir)
    except Exception:
        pass


def merge_batches_into_output(temp_dir, base_csv_path, batch_files, output_csv_path, node_count, progress_dialog=None):
    if len(batch_files) == 0:
        if os.path.exists(output_csv_path):
            os.remove(output_csv_path)
        os.rename(base_csv_path, output_csv_path)
        return output_csv_path

    current_base = base_csv_path
    pending = list(batch_files)
    pass_index = 0
    previous_group_limit = -1

    while len(pending) > 0:
        group_limit, merge_available_ram = estimate_merge_group_limit(len(pending))
        if group_limit != previous_group_limit:
            log_message(
                "Merge chunk size selected: {0} files/pass (available RAM: {1}).".format(
                    group_limit,
                    format_available_ram_text(merge_available_ram),
                ),
                progress_dialog,
            )
            previous_group_limit = group_limit

        apply_low_ram_backpressure(
            progress_dialog,
            "before merge pass {0}".format(pass_index + 1),
        )

        pass_index += 1
        group = pending[:group_limit]
        pending = pending[group_limit:]
        remaining_passes = int(math.ceil(float(len(pending)) / float(max(group_limit, 1))))
        total_passes = pass_index + remaining_passes

        next_base = os.path.join(temp_dir, "merge_pass_{0:03d}.csv".format(pass_index))
        msg = "Merging temporary CSVs (pass {0}/{1}, files in pass: {2})".format(
            pass_index, total_passes, len(group)
        )
        log_message(msg, progress_dialog)
        if progress_dialog is not None:
            progress_dialog.update_progress(msg)

        merge_pass(
            current_base,
            group,
            next_base,
            node_count,
            pass_index,
            total_passes,
            progress_dialog,
        )

        if current_base != base_csv_path:
            safe_delete_file(current_base)
        else:
            safe_delete_file(base_csv_path)
        for fp in group:
            safe_delete_file(fp)
        current_base = next_base
        force_memory_cleanup()

    if os.path.exists(output_csv_path):
        os.remove(output_csv_path)
    os.rename(current_base, output_csv_path)
    return output_csv_path


def append_suffix_before_extension(file_name, suffix_text):
    root, ext = os.path.splitext(file_name)
    return "{0}{1}{2}".format(root, suffix_text, ext)


def create_job_output_folder(base_output_dir, job_id):
    base_name = "modal_tensor_output_job_{0}".format(job_id)
    candidate = os.path.join(base_output_dir, base_name)

    if not os.path.exists(candidate):
        os.makedirs(candidate)
        return candidate

    run_idx = 2
    while True:
        candidate_name = "modal_tensor_output_run{0}_job_{1}".format(run_idx, job_id)
        candidate = os.path.join(base_output_dir, candidate_name)
        if not os.path.exists(candidate):
            os.makedirs(candidate)
            return candidate
        run_idx += 1


def write_job_metadata_file(
    job_output_dir,
    job_record,
    mode_count,
    started_epoch,
    completed_epoch,
    status_text,
    output_csv_paths,
    available_ram_start=-1,
    error_text="",
):
    if job_output_dir is None:
        return

    metadata_path = os.path.join(job_output_dir, "job_metadata.txt")
    elapsed_seconds = max(0.0, float(completed_epoch - started_epoch))
    base_output_dir = str(job_record.get("base_output_dir", "")).strip()
    if base_output_dir == "":
        base_output_dir = "(analysis WorkingDir)"

    result_labels = get_result_kind_labels(job_record.get("result_kinds", []))
    lines = [
        "Job ID: {0}".format(job_record.get("job_id", "")),
        "Status: {0}".format(status_text),
        "Modal Analysis Environment: {0}".format(job_record.get("analysis_name", "")),
        "Nodal Named Selection: {0}".format(job_record.get("named_selection_name", "")),
        "Requested Output Types: {0}".format("; ".join(result_labels)),
        "No Graphics Preference Enabled: {0}".format(format_yes_no(job_record.get("use_no_graphics", False))),
        "Mode Count: {0}".format(mode_count),
        "Available RAM at Job Start: {0}".format(format_available_ram_text(available_ram_start)),
        "Job Start Time: {0}".format(format_timestamp(started_epoch)),
        "Job Completion Time: {0}".format(format_timestamp(completed_epoch)),
        "Job Duration (s): {0:.3f}".format(elapsed_seconds),
        "Base Output Folder: {0}".format(base_output_dir),
        "Output Folder: {0}".format(job_output_dir),
        "Output CSV Files:",
    ]

    if output_csv_paths is not None and len(output_csv_paths) > 0:
        for path in output_csv_paths:
            lines.append(path)
    else:
        lines.append("(none)")

    if error_text is not None and str(error_text).strip() != "":
        lines.append("Error: {0}".format(str(error_text)))

    with open(metadata_path, "wb") as f:
        for line in lines:
            f.write(str(line))
            f.write("\n")


def run_extraction(
    solution,
    named_selection,
    result_kind,
    mode_ids,
    progress_dialog=None,
    result_kind_label=None,
    output_dir=None,
    output_name_suffix="",
):
    start_time = time.time()
    temp_objects = []
    temp_dir = None

    component_names = []
    component_objects = []
    batch_files = []
    node_ids = []
    base_csv_path = None
    if result_kind_label is None:
        result_kind_label = RESULT_KIND_LABEL_BY_ID.get(result_kind, result_kind)

    try:
        msg = "Creating temporary result objects..."
        log_message(msg, progress_dialog)
        if progress_dialog is not None:
            progress_dialog.update_progress(msg, 0, 1000)

        with begin_graphics_suspend():
            component_names, component_objects = create_component_objects(solution, named_selection, result_kind)
            temp_objects = component_objects
            comp_count = len(component_objects)
            if comp_count == 0:
                raise Exception("No temporary extraction result objects were created.")

            # Evaluate first component for first mode to establish node ordering and unit.
            first_mode = int(mode_ids[0])
            with begin_transaction(True):
                component_objects[0].SetNumber = first_mode
            component_objects[0].EvaluateAllResults()
            node_ids = extract_node_ids_from_result(component_objects[0])
            if len(node_ids) == 0:
                raise Exception("No nodes found in extracted scope.")
            first_unit = get_values_unit(component_objects[0])
            try:
                component_objects[0].ClearGeneratedData()
            except Exception:
                pass

            scale_factors = get_scale_factors(result_kind, comp_count, first_unit)
            if result_kind == RESULT_KIND_STRESS and scale_factors[0] == 1e-6:
                log_message("Stress unit '{0}' detected. Exporting stresses in MPa.".format(first_unit), progress_dialog)
            if result_kind == RESULT_KIND_DEFORMATION and scale_factors[0] == 1e3:
                log_message(
                    "Deformation unit '{0}' detected. Exporting deformations in mm.".format(first_unit),
                    progress_dialog,
                )

            msg = "Extracting coordinates for {0} scoped nodes...".format(len(node_ids))
            log_message(msg, progress_dialog)
            if progress_dialog is not None:
                progress_dialog.update_progress(msg, 5, 1000)
            node_coords = get_node_coordinates(node_ids, progress_dialog)

            temp_dir = os.path.join(
                solution.WorkingDir,
                "_tensor_extract_tmp_{0}".format(int(time.time())),
            )
            if not os.path.isdir(temp_dir):
                os.makedirs(temp_dir)

            base_csv_path = os.path.join(temp_dir, "base_coords.csv")
            write_base_coordinate_csv(base_csv_path, node_ids, node_coords)
            del node_coords
            force_memory_cleanup()

            use_solution_eval = should_use_solution_evaluate(solution, len(component_objects))
            if result_kind == RESULT_KIND_FORCE_MOMENT:
                # ENFO/ENMO are treated as a coupled channel set for this workflow.
                use_solution_eval = True
                log_message(
                    "ENFO/ENMO extraction uses solution.EvaluateAllResults() per mode; failure is treated as unavailable data for that mode/scope.",
                    progress_dialog,
                )
            if use_solution_eval:
                log_message("Using solution.EvaluateAllResults() once per mode.", progress_dialog)
            else:
                log_message("Using per-object EvaluateAllResults() (avoids evaluating all existing tree results).", progress_dialog)

            total_modes = len(mode_ids)
            mode_cursor = 0
            batch_index = 0
            while mode_cursor < total_modes:
                remaining = total_modes - mode_cursor
                batch_size, available_ram = estimate_mode_batch_size(len(node_ids), comp_count, remaining)
                batch_modes = mode_ids[mode_cursor: mode_cursor + batch_size]
                batch_index += 1

                if available_ram > 0:
                    ram_text = "{0:.2f} GiB".format(bytes_to_gib(available_ram))
                else:
                    ram_text = "unknown"
                msg = "Batch {0}: modes {1}-{2} ({3} modes), available RAM: {4}".format(
                    batch_index,
                    batch_modes[0],
                    batch_modes[-1],
                    len(batch_modes),
                    ram_text,
                )
                log_message(msg, progress_dialog)

                for local_idx in xrange(len(batch_modes)):
                    mode = int(batch_modes[local_idx])
                    global_mode_idx = mode_cursor + local_idx + 1
                    apply_low_ram_backpressure(
                        progress_dialog,
                        "before extracting mode {0}".format(mode),
                    )
                    mode_msg = "Extracting [{0}] mode {1}/{2} (set {3})".format(
                        result_kind_label,
                        global_mode_idx,
                        total_modes,
                        mode,
                    )
                    log_message(mode_msg, progress_dialog)
                    if progress_dialog is not None:
                        # reserve 80% of progress range to mode extraction
                        mode_progress = int((float(global_mode_idx) / float(total_modes)) * 800.0)
                        progress_dialog.update_progress(mode_msg, mode_progress, 1000)

                    with begin_transaction(True):
                        for obj in component_objects:
                            obj.SetNumber = mode

                    if use_solution_eval:
                        try:
                            solution.EvaluateAllResults()
                        except Exception as eval_all_ex:
                            if result_kind == RESULT_KIND_FORCE_MOMENT:
                                raise Exception(
                                    "ENFO/ENMO evaluation failed at mode {0}. "
                                    "Requested scope appears unavailable in RST for force/moment channels. Detail: {1}".format(
                                        mode, str(eval_all_ex)
                                    )
                                )
                            else:
                                raise
                    else:
                        for eval_idx in xrange(comp_count):
                            obj = component_objects[eval_idx]
                            obj.EvaluateAllResults()

                    mode_component_values = []
                    first_component_values, first_is_base_order = extract_aligned_values(
                        component_objects[0],
                        node_ids,
                        scale_factors[0],
                        return_order_match=True,
                    )
                    mode_component_values.append(first_component_values)

                    for comp_idx in xrange(1, comp_count):
                        obj = component_objects[comp_idx]
                        if first_is_base_order:
                            values = extract_scaled_values(obj, scale_factors[comp_idx])
                        else:
                            values = extract_aligned_values(obj, node_ids, scale_factors[comp_idx])
                        mode_component_values.append(values)

                    mode_file = os.path.join(temp_dir, "mode_{0:05d}.csv".format(mode))
                    write_mode_csv(mode_file, mode, component_names, node_ids, mode_component_values)
                    batch_files.append(mode_file)

                    for obj in component_objects:
                        try:
                            obj.ClearGeneratedData()
                        except Exception:
                            pass

                    del mode_component_values
                    if global_mode_idx % 2 == 0:
                        force_memory_cleanup(True)
                    else:
                        force_memory_cleanup()
                    apply_low_ram_backpressure(
                        progress_dialog,
                        "after extracting mode {0}".format(mode),
                    )

                mode_cursor += len(batch_modes)
                force_memory_cleanup()

        output_name = RESULT_OUTPUT_FILES[result_kind]
        if output_name_suffix is not None and str(output_name_suffix).strip() != "":
            output_name = append_suffix_before_extension(output_name, str(output_name_suffix))

        if output_dir is None or str(output_dir).strip() == "":
            output_dir = solution.WorkingDir
        if not os.path.isdir(output_dir):
            os.makedirs(output_dir)
        output_csv_path = os.path.join(output_dir, output_name)

        msg = "Merging temporary CSV chunks into final output..."
        log_message(msg, progress_dialog)
        if progress_dialog is not None:
            progress_dialog.update_progress(msg, 820, 1000)

        merge_batches_into_output(
            temp_dir=temp_dir,
            base_csv_path=base_csv_path,
            batch_files=batch_files,
            output_csv_path=output_csv_path,
            node_count=len(node_ids),
            progress_dialog=progress_dialog,
        )

        if progress_dialog is not None:
            progress_dialog.update_progress("Completed.", 1000, 1000)

        elapsed = time.time() - start_time
        log_message("Extraction completed in {0:.1f} s".format(elapsed), progress_dialog)
        log_message("CSV file saved at: {0}".format(output_csv_path), progress_dialog)
        return output_csv_path

    finally:
        cleanup_component_objects(temp_objects)
        cleanup_temp_folder(temp_dir)
        restore_ui_interaction_state()
        force_memory_cleanup(True)


def main():
    modal_analyses = get_modal_analysis_candidates()
    if len(modal_analyses) == 0:
        raise Exception("No modal analysis environment found in Mechanical tree.")

    nodal_named_selections = get_nodal_named_selections()
    if len(nodal_named_selections) == 0:
        raise Exception("No named selections found in tree.")

    def run_from_dialog(dialog):
        pending_jobs = dialog.pending_jobs()
        if len(pending_jobs) == 0:
            raise Exception("No pending jobs found. Use 'Submit Jobs' before running.")

        total_jobs = len(pending_jobs)
        log_message("Starting extraction queue with {0} pending jobs.".format(total_jobs), dialog)
        log_message(
            "Jobs are processed sequentially with per-kind and per-job cleanup to keep RAM usage stable.",
            dialog,
        )

        completed_job_ids = []
        failed_jobs = []
        all_output_csvs = []

        for job_idx in xrange(total_jobs):
            job = pending_jobs[job_idx]
            job_id = int(job.get("job_id", job_idx + 1))
            selected_analysis = job.get("analysis", None)
            selected_ns = job.get("named_selection", None)
            selected_kinds = list(job.get("result_kinds", []))
            use_no_graphics_pref = bool(job.get("use_no_graphics", False))

            if selected_analysis is None or selected_ns is None or len(selected_kinds) == 0:
                dialog.set_job_status(job_id, JOB_STATUS_FAILED)
                cfg_err = "Job configuration is incomplete."
                job["error"] = cfg_err
                failed_jobs.append((job_id, cfg_err))
                log_message("Job {0} failed before start: {1}".format(job_id, cfg_err), dialog)
                continue

            pref_state = None
            solution = None
            mode_ids = []
            output_dir = None
            output_csvs = []
            job_start = time.time()
            job_end = job_start
            available_ram_start = get_available_ram_bytes()

            job["started_epoch"] = job_start
            job["completed_epoch"] = 0.0
            job["error"] = ""
            dialog.set_job_status(job_id, JOB_STATUS_RUNNING)

            try:
                solution, mode_ids = get_analysis_solution_and_modes(selected_analysis)
                selected_base_output_dir = str(job.get("base_output_dir", "")).strip()
                if selected_base_output_dir == "":
                    selected_base_output_dir = solution.WorkingDir
                if not os.path.isdir(selected_base_output_dir):
                    os.makedirs(selected_base_output_dir)

                job["base_output_dir"] = selected_base_output_dir
                output_dir = create_job_output_folder(selected_base_output_dir, job_id)
                job["output_dir"] = output_dir

                kind_labels = get_result_kind_labels(selected_kinds)
                start_msg = (
                    "Job {0}/{1} (ID {2}) | Analysis: {3} | NS: {4} | Result types: {5} | Modes: 1..{6}"
                ).format(
                    job_idx + 1,
                    total_jobs,
                    job_id,
                    selected_analysis.Name,
                    selected_ns.Name,
                    "; ".join(kind_labels),
                    len(mode_ids),
                )
                log_message(start_msg, dialog)
                log_message("Job {0} base output folder: {1}".format(job_id, selected_base_output_dir), dialog)
                log_message("Job {0} output folder: {1}".format(job_id, output_dir), dialog)
                dialog.update_progress(start_msg, 0, 1000)

                if use_no_graphics_pref:
                    pref_state = enter_no_graphics_preference_mode(dialog)
                else:
                    log_message("Job {0}: No-Graphics preference is disabled.".format(job_id), dialog)

                n_selected = len(selected_kinds)
                for run_idx in xrange(n_selected):
                    selected_kind = selected_kinds[run_idx]
                    kind_label = RESULT_KIND_LABEL_BY_ID.get(selected_kind, selected_kind)
                    available_ram = get_available_ram_bytes()
                    stage_msg = (
                        "Job {0}/{1} (ID {2}) | Extraction {3}/{4}: {5} | Available RAM before start: {6}"
                    ).format(
                        job_idx + 1,
                        total_jobs,
                        job_id,
                        run_idx + 1,
                        n_selected,
                        kind_label,
                        format_available_ram_text(available_ram),
                    )
                    log_message(stage_msg, dialog)
                    dialog.update_progress(stage_msg, 0, 1000)

                    output_csv = run_extraction(
                        solution,
                        selected_ns,
                        selected_kind,
                        mode_ids,
                        dialog,
                        kind_label,
                        output_dir=output_dir,
                        output_name_suffix="_job_{0}".format(job_id),
                    )
                    output_csvs.append(output_csv)
                    all_output_csvs.append(output_csv)

                    # Explicit cleanup between result kinds for stable memory behavior.
                    force_memory_cleanup(True)
                    time.sleep(0.1)

                job_end = time.time()
                job["completed_epoch"] = job_end
                job["output_csv_paths"] = list(output_csvs)
                dialog.set_job_status(job_id, JOB_STATUS_COMPLETED)
                completed_job_ids.append(job_id)

                write_job_metadata_file(
                    job_output_dir=output_dir,
                    job_record=job,
                    mode_count=len(mode_ids),
                    started_epoch=job_start,
                    completed_epoch=job_end,
                    status_text=JOB_STATUS_COMPLETED,
                    output_csv_paths=output_csvs,
                    available_ram_start=available_ram_start,
                )

                elapsed = job_end - job_start
                log_message("Job {0} completed in {1:.1f} s".format(job_id, elapsed), dialog)

            except Exception as job_ex:
                job_end = time.time()
                job_err = str(job_ex)
                job["completed_epoch"] = job_end
                job["output_csv_paths"] = list(output_csvs)
                job["error"] = job_err
                dialog.set_job_status(job_id, JOB_STATUS_FAILED)
                failed_jobs.append((job_id, job_err))
                log_message("Job {0} failed: {1}".format(job_id, job_err), dialog)

                if output_dir is not None:
                    write_job_metadata_file(
                        job_output_dir=output_dir,
                        job_record=job,
                        mode_count=len(mode_ids),
                        started_epoch=job_start,
                        completed_epoch=job_end,
                        status_text=JOB_STATUS_FAILED,
                        output_csv_paths=output_csvs,
                        available_ram_start=available_ram_start,
                        error_text=job_err,
                    )
            finally:
                try:
                    restore_no_graphics_preference_mode(pref_state, dialog)
                except Exception:
                    pass

                # Release references and force collection between jobs.
                solution = None
                mode_ids = []
                force_memory_cleanup(True)
                time.sleep(0.1)
                restore_ui_interaction_state()

        summary_lines = [
            "Extraction jobs completed.",
            "Completed jobs: {0}".format(len(completed_job_ids)),
            "Failed jobs: {0}".format(len(failed_jobs)),
        ]

        if len(all_output_csvs) > 0:
            summary_lines.append("")
            summary_lines.append("Output CSV files:")
            for out_idx in xrange(len(all_output_csvs)):
                out_path = all_output_csvs[out_idx]
                summary_lines.append("{0}. {1}".format(out_idx + 1, out_path))
                summary_lines.append("")

        if len(failed_jobs) > 0:
            summary_lines.append("")
            summary_lines.append("Failed job details:")
            for failed_job_id, failed_err in failed_jobs:
                summary_lines.append("Job {0}: {1}".format(failed_job_id, failed_err))

        MessageBox.Show(
            dialog,
            "\n".join(summary_lines),
            "Modal Extractor",
            MessageBoxButtons.OK,
            MessageBoxIcon.Information if len(failed_jobs) == 0 else MessageBoxIcon.Warning,
        )

    dlg = ExtractOptionsDialog(modal_analyses, nodal_named_selections, run_from_dialog)
    dlg.ShowDialog()
    log_message("Modal extractor dialog closed by user.")


main()
