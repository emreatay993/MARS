def get_app_preference(pref_id, default_value=None):
    try:
        pref_val = ExtAPI.Application.GetPreference(pref_id)
        if pref_val is None:
            return default_value
        pref_str = str(pref_val)
        if pref_str == "":
            return default_value
        return pref_str
    except Exception:
        pass

    # Fallback through Mechanical's jscript bridge.
    try:
        cmd = "returnFromScript(DS.Script.wb.PreferenceMgr.Preference('{0}'));".format(pref_id)
        pref_val = ExtAPI.Application.ScriptByName("jscript").ExecuteCommand(cmd)
        if pref_val is None:
            return default_value
        pref_str = str(pref_val)
        if pref_str == "":
            return default_value
        return pref_str
    except Exception:
        return default_value


def set_app_preference(pref_id, pref_value):
    value_str = str(pref_value)

    # First try direct API, if available.
    try:
        ExtAPI.Application.SetPreference(pref_id, value_str)
        return True
    except Exception:
        pass

    # Fallback through jscript/PreferenceMgr (used internally by Mechanical UI scripts).
    try:
        cmd = (
            "DS.Script.wb.PreferenceMgr.Preference('{0}') = '{1}';"
            "DS.Script.wb.PreferenceMgr.Save();"
            "returnFromScript(DS.Script.wb.PreferenceMgr.Preference('{0}'));"
        ).format(pref_id, value_str)
        new_val = ExtAPI.Application.ScriptByName("jscript").ExecuteCommand(cmd)
        if new_val is None:
            return True
        return str(new_val) == value_str
    except Exception:
        return False


def enter_no_graphics_preference_mode(progress_dialog=None):
    original_val = get_app_preference(PREF_LEVEL_OF_DETAIL, None)
    changed_ok = set_app_preference(PREF_LEVEL_OF_DETAIL, PREF_LEVEL_OF_DETAIL_NO_GRAPHICS)

    if changed_ok:
        log_message(
            "Set preference {0}=0 (No Graphics) for extraction.".format(PREF_LEVEL_OF_DETAIL),
            progress_dialog,
        )
    else:
        log_message(
            "Could not set {0}=0 via API in this session; using suspend-based redraw suppression only.".format(
                PREF_LEVEL_OF_DETAIL
            ),
            progress_dialog,
        )

    return {
        "pref_id": PREF_LEVEL_OF_DETAIL,
        "original_value": original_val,
        "restore_needed": (original_val is not None and str(original_val) != PREF_LEVEL_OF_DETAIL_NO_GRAPHICS),
        "set_succeeded": changed_ok,
    }


def restore_no_graphics_preference_mode(pref_state, progress_dialog=None):
    if pref_state is None:
        return
    if not pref_state.get("restore_needed", False):
        return

    pref_id = pref_state.get("pref_id", PREF_LEVEL_OF_DETAIL)
    original_val = pref_state.get("original_value", None)
    if original_val is None:
        return

    restored_ok = set_app_preference(pref_id, str(original_val))
    if restored_ok:
        log_message(
            "Restored preference {0} back to {1}.".format(pref_id, str(original_val)),
            progress_dialog,
        )
    else:
        log_message(
            "Warning: could not restore {0} to {1}. Please check Mechanical Options > Graphics.".format(
                pref_id, str(original_val)
            ),
            progress_dialog,
        )


def normalize_unit_system_value(unit_system_value):
    if unit_system_value is None:
        return None

    for attr_name in ("Value", "value__"):
        try:
            attr_value = getattr(unit_system_value, attr_name)
            return int(attr_value)
        except Exception:
            pass

    try:
        return int(unit_system_value)
    except Exception:
        pass

    return None


def format_unit_system_value(unit_system_value):
    if unit_system_value is None:
        return "unknown"

    try:
        text = str(unit_system_value)
        if text is not None and text.strip() != "":
            return text
    except Exception:
        pass

    normalized = normalize_unit_system_value(unit_system_value)
    if normalized is None:
        return "unknown"
    return "ID {0}".format(normalized)


def resolve_standard_nmm_unit_system_for_application():
    candidate_names = ["MechanicalUnitSystem", "MechanicalUnitSystemEnum"]
    for type_name in candidate_names:
        enum_type = globals().get(type_name, None)
        if enum_type is None:
            continue
        try:
            return getattr(enum_type, "StandardNMM")
        except Exception:
            pass
    return UNIT_SYSTEM_STANDARD_NMM_VALUE

    candidate_names = ["WBUnitSystemType", "MechanicalUnitSystem", "MechanicalUnitSystemEnum"]
    for type_name in candidate_names:
        enum_type = globals().get(type_name, None)
        if enum_type is None:
            continue
        try:
            return getattr(enum_type, "StandardNMM")
        except Exception:
            pass
    return UNIT_SYSTEM_STANDARD_NMM_VALUE


def enter_standard_nmm_unit_system(progress_dialog=None):
    original_active = None
    original_project = None

    try:
        original_active = ExtAPI.Application.ActiveUnitSystem
    except Exception:
        pass

    try:
        original_project = ExtAPI.DataModel.Project.UnitSystem
    except Exception:
        pass

    target_app_unit = resolve_standard_nmm_unit_system_for_application()
    target_project_unit = resolve_standard_nmm_unit_system_for_project()
    app_set_error = None
    project_set_error = None

    try:
        ExtAPI.Application.ActiveUnitSystem = target_app_unit
    except Exception as ex:
        app_set_error = ex

    try:
        ExtAPI.DataModel.Project.UnitSystem = target_project_unit
    except Exception as ex:
        project_set_error = ex

    current_active = None
    current_project = None
    try:
        current_active = ExtAPI.Application.ActiveUnitSystem
    except Exception:
        pass

    try:
        current_project = ExtAPI.DataModel.Project.UnitSystem
    except Exception:
        pass

    active_ok = normalize_unit_system_value(current_active) == UNIT_SYSTEM_STANDARD_NMM_VALUE
    project_ok = normalize_unit_system_value(current_project) == UNIT_SYSTEM_STANDARD_NMM_VALUE
    if not active_ok and not project_ok:
        detail_parts = []
        if app_set_error is not None:
            detail_parts.append("ActiveUnitSystem failed: {0}".format(str(app_set_error)))
        if project_set_error is not None:
            detail_parts.append("Project.UnitSystem failed: {0}".format(str(project_set_error)))
        if len(detail_parts) == 0:
            detail_parts.append(
                "Current ActiveUnitSystem={0}, Project.UnitSystem={1}".format(
                    format_unit_system_value(current_active),
                    format_unit_system_value(current_project),
                )
            )
        raise Exception(
            "Could not enforce Mechanical unit system to {0}. {1}".format(
                UNIT_SYSTEM_STANDARD_NMM_LABEL,
                " | ".join(detail_parts),
            )
        )

    log_message(
        "Forced Mechanical unit system to {0}. ActiveUnitSystem={1}, Project.UnitSystem={2}.".format(
            UNIT_SYSTEM_STANDARD_NMM_LABEL,
            format_unit_system_value(current_active),
            format_unit_system_value(current_project),
        ),
        progress_dialog,
    )

    return {
        "original_active": original_active,
        "original_project": original_project,
        "restore_active": (
            original_active is not None
            and normalize_unit_system_value(original_active) != normalize_unit_system_value(current_active)
        ),
        "restore_project": (
            original_project is not None
            and normalize_unit_system_value(original_project) != normalize_unit_system_value(current_project)
        ),
    }

        try:
            ExtAPI.DataModel.Project.UnitSystem = unit_state.get("original_project", None)
        except Exception as ex:
            project_restore_error = ex

    current_active = None
    current_project = None
    try:
        current_active = ExtAPI.Application.ActiveUnitSystem
    except Exception:
        pass

    try:
        current_project = ExtAPI.DataModel.Project.UnitSystem
    except Exception:
        pass

    if active_restore_error is None and project_restore_error is None:
        log_message(
            "Restored Mechanical unit system. ActiveUnitSystem={0}, Project.UnitSystem={1}.".format(
                format_unit_system_value(current_active),
                format_unit_system_value(current_project),
            ),

    if active_restore_error is not None:
        warning_parts.append("ActiveUnitSystem restore failed: {0}".format(str(active_restore_error)))
    if project_restore_error is not None:
        warning_parts.append("Project.UnitSystem restore failed: {0}".format(str(project_restore_error)))
    log_message(
        "Warning: could not fully restore Mechanical unit system. {0}".format(" | ".join(warning_parts)),
        progress_dialog,
    )

def _probe_available_ram_bytes():
    """Actual OS query for available RAM (no caching)."""
    global _AVAILABLE_MB_COUNTER
    global _AVAILABLE_MB_COUNTER_LAST_INIT

    try:
        from Microsoft.VisualBasic.Devices import ComputerInfo
        return int(ComputerInfo().AvailablePhysicalMemory)
    except Exception:
        pass

    try:
        clr.AddReference("System")
        from System.Diagnostics import PerformanceCounter
        now = time.time()
        should_recreate_counter = (
            _AVAILABLE_MB_COUNTER is None
            or (RAM_PROBE_PERF_COUNTER_REFRESH_SEC > 0 and (now - _AVAILABLE_MB_COUNTER_LAST_INIT) > RAM_PROBE_PERF_COUNTER_REFRESH_SEC)
        )
        if should_recreate_counter:
            try_dispose_dotnet_object(_AVAILABLE_MB_COUNTER)
            _AVAILABLE_MB_COUNTER = PerformanceCounter("Memory", "Available MBytes")
            _AVAILABLE_MB_COUNTER_LAST_INIT = now
            try:
                _AVAILABLE_MB_COUNTER.NextValue()
            except Exception:
                pass

        mb = float(_AVAILABLE_MB_COUNTER.NextValue())
        return int(mb * 1024 * 1024)
    except Exception:
        try_dispose_dotnet_object(_AVAILABLE_MB_COUNTER)
        _AVAILABLE_MB_COUNTER = None
        _AVAILABLE_MB_COUNTER_LAST_INIT = 0.0

    searcher = None
    wmi_rows = None
    try:
        clr.AddReference("System.Management")
        from System.Management import ManagementObjectSearcher
        searcher = ManagementObjectSearcher("SELECT FreePhysicalMemory FROM Win32_OperatingSystem")
        wmi_rows = searcher.Get()
        for mo in wmi_rows:
            free_kb = int(mo["FreePhysicalMemory"])
            return int(free_kb * 1024)
    except Exception:
        pass
    finally:
        try_dispose_dotnet_object(wmi_rows)
        try_dispose_dotnet_object(searcher)

    return -1


def get_available_ram_bytes():
    global _AVAILABLE_RAM_BYTES_CACHE
    global _AVAILABLE_RAM_BYTES_LAST_REFRESH

    now = time.time()
    if (
        _AVAILABLE_RAM_BYTES_CACHE > 0
        and RAM_AVAILABLE_CACHE_COOLDOWN_SEC > 0
        and (now - _AVAILABLE_RAM_BYTES_LAST_REFRESH) <= RAM_AVAILABLE_CACHE_COOLDOWN_SEC
    ):
        return _AVAILABLE_RAM_BYTES_CACHE

    result = _probe_available_ram_bytes()
    if result > 0:
        _AVAILABLE_RAM_BYTES_CACHE = result
        _AVAILABLE_RAM_BYTES_LAST_REFRESH = now
    return result


def get_total_ram_bytes():
    global _TOTAL_RAM_BYTES_CACHE
    global _TOTAL_RAM_BYTES_LAST_REFRESH

    now = time.time()
    if (
        _TOTAL_RAM_BYTES_CACHE > 0
        and TOTAL_RAM_CACHE_REFRESH_SEC > 0
        and (now - _TOTAL_RAM_BYTES_LAST_REFRESH) <= TOTAL_RAM_CACHE_REFRESH_SEC
    ):
        return _TOTAL_RAM_BYTES_CACHE

    total_bytes = -1
    try:
        from Microsoft.VisualBasic.Devices import ComputerInfo
        total_bytes = int(ComputerInfo().TotalPhysicalMemory)
    except Exception:
        pass

    if total_bytes <= 0:
        searcher = None
        wmi_rows = None
        try:
            clr.AddReference("System.Management")
            from System.Management import ManagementObjectSearcher
            searcher = ManagementObjectSearcher("SELECT TotalVisibleMemorySize FROM Win32_OperatingSystem")
            wmi_rows = searcher.Get()
            for mo in wmi_rows:
                total_kb = int(mo["TotalVisibleMemorySize"])
                total_bytes = int(total_kb * 1024)
                break
        except Exception:
            pass
        finally:
            try_dispose_dotnet_object(wmi_rows)
            try_dispose_dotnet_object(searcher)

    if total_bytes > 0:
        _TOTAL_RAM_BYTES_CACHE = total_bytes
        _TOTAL_RAM_BYTES_LAST_REFRESH = now

    return total_bytes


def get_available_ram_percent(available_bytes=None, total_bytes=None):
    if available_bytes is None or available_bytes <= 0:
        available_bytes = get_available_ram_bytes()
    if total_bytes is None or total_bytes <= 0:
        total_bytes = get_total_ram_bytes()

    if available_bytes > 0 and total_bytes > 0:
        pct = (float(available_bytes) * 100.0) / float(total_bytes)
        if pct < 0.0:
            pct = 0.0
        if pct > 100.0:
            pct = 100.0
        return pct, total_bytes

    return -1.0, total_bytes


def is_run_cancel_requested(progress_dialog=None):
    if progress_dialog is None:
        return False
    checker = getattr(progress_dialog, "is_cancel_requested", None)
    if checker is None:
        return False
    try:
        return bool(checker())
    except Exception:
        return False


def raise_if_run_cancel_requested(progress_dialog=None, context_text=""):
    if not is_run_cancel_requested(progress_dialog):
        return
    msg = "Extraction was cancelled by user request."
    if context_text is not None and str(context_text).strip() != "":
        msg += " Context: {0}".format(str(context_text))
    raise UserCancelledException(msg)


def get_ram_strategy(preset_key=None):
    if preset_key is not None and preset_key in RAM_STRATEGY_PRESETS:
        return dict(RAM_STRATEGY_PRESETS[preset_key])
    return dict(RAM_STRATEGY_PRESETS[RAM_STRATEGY_DEFAULT])


def get_job_ram_strategy(job_record):
    if job_record is None:
        return get_ram_strategy(RAM_STRATEGY_DEFAULT)

    if "ram_strategy" in job_record:
        key = str(job_record.get("ram_strategy", RAM_STRATEGY_DEFAULT)).strip().lower()
        return get_ram_strategy(key)

    # Legacy compatibility: map old "low_ram_free_percent" to nearest preset.
    if "low_ram_free_percent" in job_record:
        try:
            pct = float(job_record.get("low_ram_free_percent"))
        except Exception:
            return get_ram_strategy(RAM_STRATEGY_DEFAULT)
        if pct >= 45.0:
            return get_ram_strategy("conservative")
        elif pct >= 28.0:
            return get_ram_strategy("balanced")
        else:
            return get_ram_strategy("aggressive")

    if "low_ram_used_percent" in job_record:
        try:
            legacy_used_pct = float(job_record.get("low_ram_used_percent"))
            free_pct = 100.0 - legacy_used_pct
        except Exception:
            return get_ram_strategy(RAM_STRATEGY_DEFAULT)
        if free_pct >= 45.0:
            return get_ram_strategy("conservative")
        elif free_pct >= 28.0:
            return get_ram_strategy("balanced")
        else:
            return get_ram_strategy("aggressive")

    return get_ram_strategy(RAM_STRATEGY_DEFAULT)


def evaluate_low_ram_condition(ram_strategy=None, available_bytes=None, total_bytes=None):
    if ram_strategy is None:
        ram_strategy = get_ram_strategy(RAM_STRATEGY_DEFAULT)
    # Accept either a strategy dict or a raw float for backward compatibility (merge functions).
    if isinstance(ram_strategy, (int, float)):
        threshold_percent = float(ram_strategy)
    else:
        threshold_percent = float(ram_strategy.get("low_ram_free_percent", 35.0))
    if available_bytes is None or available_bytes <= 0:
        available_bytes = get_available_ram_bytes()
    available_percent, total_bytes = get_available_ram_percent(available_bytes, total_bytes)

    if available_percent >= 0.0:
        return {
            "triggered": (available_percent <= threshold_percent),
            "threshold_percent": threshold_percent,
            "available_percent": available_percent,
            "available_bytes": available_bytes,
            "total_bytes": total_bytes,
            "percent_mode": True,
            "fallback_bytes": LOW_RAM_BACKPRESSURE_FALLBACK_BYTES,
        }

    return {
        "triggered": (available_bytes > 0 and available_bytes < LOW_RAM_BACKPRESSURE_FALLBACK_BYTES),
        "threshold_percent": threshold_percent,
        "available_percent": -1.0,
        "available_bytes": available_bytes,
        "total_bytes": total_bytes,
        "percent_mode": False,
        "fallback_bytes": LOW_RAM_BACKPRESSURE_FALLBACK_BYTES,
    }


def force_memory_cleanup(wait_for_finalizers=False):
    try:
        gc.collect()
    except Exception:
        pass

    # Mechanical objects and arrays are .NET-backed; collect both runtimes.
    try:
        clr.AddReference("System")
        from System import GC as DotNetGC
        if wait_for_finalizers:
            try:
                from System.Runtime import GCSettings, GCLargeObjectHeapCompactionMode
                GCSettings.LargeObjectHeapCompactionMode = GCLargeObjectHeapCompactionMode.CompactOnce
            except Exception:
                pass
        DotNetGC.Collect()
        if wait_for_finalizers:
            DotNetGC.WaitForPendingFinalizers()
            DotNetGC.Collect()
    except Exception:
        pass


def apply_low_ram_backpressure(progress_dialog=None, context_text="", ram_strategy=None, low_ram_free_percent=None):
    effective_strategy = ram_strategy
    if effective_strategy is None and low_ram_free_percent is not None:
        effective_strategy = low_ram_free_percent
    ram_state = evaluate_low_ram_condition(effective_strategy)
    available_bytes = ram_state.get("available_bytes", -1)
    if ram_state.get("triggered", False):
        if ram_state.get("percent_mode", False):
            msg = "Low available RAM detected ({0} free, {1} available; trigger: <= {2:.1f}% free). Running aggressive cleanup.".format(
                format_available_ram_percent_text(ram_state.get("available_percent", -1.0)),
                format_available_ram_text(available_bytes),
                ram_state.get("threshold_percent", 35.0),
            )
        else:
            msg = "Low available RAM detected ({0} available; fallback trigger: below {1}; configured trigger: <= {2:.1f}% free, total RAM query unavailable). Running aggressive cleanup.".format(
                format_available_ram_text(available_bytes),
                format_available_ram_text(ram_state.get("fallback_bytes", LOW_RAM_BACKPRESSURE_FALLBACK_BYTES)),
                ram_state.get("threshold_percent", 35.0),
            )
        if context_text is not None and str(context_text).strip() != "":
            msg += " Context: {0}".format(str(context_text))
        log_message(msg, progress_dialog)
        force_memory_cleanup(True)
    return ram_state


def estimate_mode_batch_size(node_count, component_count, remaining_modes, ram_strategy=None):
    if remaining_modes <= 0:
        return 1, -1, False

    if ram_strategy is None:
        ram_strategy = get_ram_strategy(RAM_STRATEGY_DEFAULT)

    available_bytes = get_available_ram_bytes()
    if node_count <= 0 or component_count <= 0:
        return 1, available_bytes, False

    bytes_per_mode = int(node_count * component_count * 8.0 * OVERHEAD_MULTIPLIER)

    ram_state = evaluate_low_ram_condition(ram_strategy, available_bytes=available_bytes)
    is_low_ram = ram_state.get("triggered", False)

    fraction = RAM_SAFETY_FRACTION_LOW if is_low_ram else ram_strategy.get("safety_fraction", RAM_SAFETY_FRACTION_NORMAL)

    if available_bytes > 0:
        safe_bytes = int(max(RAM_MIN_SAFE_BYTES, available_bytes * fraction))
        estimate = int(float(safe_bytes) / float(max(bytes_per_mode, 1)))
    else:
        estimate = 1

    if estimate < 1:
        estimate = 1
    if estimate > RAM_MAX_MODES_PER_BATCH:
        estimate = RAM_MAX_MODES_PER_BATCH
    if estimate > remaining_modes:
        estimate = remaining_modes

    return estimate, available_bytes, is_low_ram


def estimate_merge_group_limit(pending_count, low_ram_free_percent=None):
    if pending_count <= 0:
        return 1, -1

    available_bytes = get_available_ram_bytes()
    if available_bytes <= 0:
        estimate = MERGE_GROUP_LIMIT_UNKNOWN_RAM
    elif evaluate_low_ram_condition(low_ram_free_percent, available_bytes=available_bytes).get("triggered", False):
        estimate = MERGE_GROUP_LIMIT_LOW_RAM
    elif available_bytes < MERGE_GROUP_MID_RAM_THRESHOLD_BYTES:
        estimate = MERGE_GROUP_LIMIT_MID_RAM
    elif available_bytes < MERGE_GROUP_HIGH_RAM_THRESHOLD_BYTES:
        estimate = MERGE_GROUP_LIMIT_HIGH_RAM
    elif available_bytes < MERGE_GROUP_VERY_HIGH_RAM_THRESHOLD_BYTES:
        estimate = MERGE_GROUP_LIMIT_VERY_HIGH_RAM
    else:
        estimate = MERGE_GROUP_LIMIT_MAX

    if estimate < MERGE_GROUP_LIMIT_MIN:
        estimate = MERGE_GROUP_LIMIT_MIN
    if estimate > MERGE_GROUP_LIMIT_MAX:
        estimate = MERGE_GROUP_LIMIT_MAX
    if estimate > pending_count:
        estimate = pending_count

    return estimate, available_bytes


def is_modal_analysis_candidate(analysis):
    if analysis is None:
        return False

    analysis_type = getattr(analysis, "AnalysisType", None)

    if analysis_type is not None:
        # Enum numeric check (Modal=5 in Mechanical AnalysisType enum).
        int_candidates = [
            analysis_type,
            getattr(analysis_type, "Value", None),
            getattr(analysis_type, "value__", None),
        ]
        for candidate in int_candidates:
            try:
                if int(candidate) == ANALYSIS_TYPE_MODAL_VALUE:
                    return True
            except Exception:
                pass

        # Enum name check across Mechanical versions.
        at_text = str(analysis_type).strip().lower()
        if at_text == "modal" or at_text.endswith(".modal"):
            return True

    # Secondary filter by runtime .NET type name.
    type_text = str(analysis.GetType().FullName).lower()
    if "modal" in type_text:
        return True

    return False


def get_modal_analysis_candidates():
    analyses = as_list(ExtAPI.DataModel.Project.Model.Analyses)
    if len(analyses) == 0:
        raise Exception("No analysis found in Mechanical tree.")

    modal_analyses = []
    for analysis in analyses:
        if is_modal_analysis_candidate(analysis):
            modal_analyses.append(analysis)

    if len(modal_analyses) == 0:
        raise Exception("No modal analysis environment was detected in tree.")
    return modal_analyses


def get_analysis_solution_and_modes(selected_analysis):
    if selected_analysis is None:
        raise Exception("No analysis selected.")

    solution = selected_analysis.Solution
    if solution is None:
        raise Exception("Selected analysis has no Solution object.")

    results_data = selected_analysis.GetResultsData()
    if results_data is None:
        raise Exception("Results data unavailable. Solve or relink result file first.")

    try:
        mode_count = len(results_data.ListTimeFreq)
    finally:
        try:
            results_data.Dispose()
        except Exception:
            pass

    if mode_count <= 0:
        raise Exception("No modal sets were found in analysis '{0}'.".format(selected_analysis.Name))

    mode_ids = range(1, mode_count + 1)
    return solution, mode_ids


def get_solution_working_directory(solution):
    if solution is None:
        return ""
    try:
        path_text = str(solution.WorkingDir).strip()
        if path_text != "":
            return path_text
    except Exception:
        pass
    return ""


def get_solution_result_file_directory(solution):
    if solution is None:
        return ""

    try:
        path_text = str(solution.ResultFileDirectory).strip()
        if path_text != "" and os.path.isdir(path_text):
            return path_text
    except Exception:
        pass

    try:
        result_path = str(solution.ResultFilePath).strip()
        if result_path != "":
            if os.path.isdir(result_path):
                return result_path
            result_parent = os.path.dirname(result_path)
            if result_parent != "" and os.path.isdir(result_parent):
                return result_parent
    except Exception:
        pass

    return ""


def get_solution_default_io_directory(solution):
    result_dir = get_solution_result_file_directory(solution)
    if result_dir != "":
        return result_dir

    working_dir = get_solution_working_directory(solution)
    if working_dir != "":
        return working_dir

    return os.getcwd()

def get_global_coordinate_system_object():
    cs_children = as_list(ExtAPI.DataModel.Project.Model.CoordinateSystems.Children)
    if len(cs_children) > 0:
        for cs in cs_children:

        return cs_children[0]

    all_cs = as_list(ExtAPI.DataModel.GetObjectsByType(DataModelObjectCategory.CoordinateSystem))
    if len(all_cs) > 0:
        for cs in all_cs:
            if bool(getattr(cs, "IsGlobal", False)):
                return cs
        return all_cs[0]

    return None


def is_nodal_named_selection(named_selection, mesh_data):
    try:
        location = named_selection.Location
        sel_type_text = str(location.SelectionType).lower().replace("_", "")
        if "meshnode" in sel_type_text:
            return True
        if "node" in sel_type_text and "mesh" in sel_type_text:
            return True

        ids = location.Ids
        if ids is None or len(ids) == 0:
            return False
        if mesh_data is None:
            return True

        probe_count = min(5, len(ids))
        hits = 0
        for i in xrange(probe_count):
            nid = int(ids[i])
            node = mesh_data.NodeById(nid)
            if node is not None:
                hits += 1
        return hits > 0
    except Exception:
        return False


def get_nodal_named_selections():
    all_ns = as_list(ExtAPI.DataModel.GetObjectsByType(DataModelObjectCategory.NamedSelection))
    all_ns_sorted = sorted(all_ns, key=lambda n: str(n.Name).lower())
    mesh_data = get_mesh_data_global()

    nodal = []
    for ns in all_ns_sorted:
        if is_nodal_named_selection(ns, mesh_data):
            nodal.append(ns)

    if len(nodal) == 0:
        # Fallback if nodal detection cannot confidently classify.
        return all_ns_sorted
    return nodal

    def _refresh_mode_limit_control(self, force_to_max=False):
        if not hasattr(self, "mode_limit_spinner"):
            return

        mode_count = 1
        analysis_obj = self.selected_analysis()
        if analysis_obj is not None:
            try:
                _, mode_ids = get_analysis_solution_and_modes(analysis_obj)
                mode_count = len(mode_ids)
            except Exception:
                mode_count = 1
        if mode_count < 1:
            mode_count = 1

        try:
            current_value = int(self.mode_limit_spinner.Value)
        except Exception:
            current_value = mode_count

        self.mode_limit_spinner.Minimum = 1
        self.mode_limit_spinner.Maximum = mode_count
        if force_to_max or current_value < 1 or current_value > mode_count:
            current_value = mode_count
        self.mode_limit_spinner.Value = current_value

    def _make_job_record(self):
        selected_analysis = self.selected_analysis()
        selected_ns = self.selected_named_selection()
        selected_kinds = self.selected_result_kinds()
        selected_base_output_dir = self.selected_base_output_dir()
        selected_mode_limit = None

        if selected_analysis is None:
            self._show_warning("Please select a modal analysis environment.", "Missing Analysis")
            return None
        if selected_ns is None:
            self._show_warning("Please select a nodal named selection.", "Missing Selection")
            return None
        if len(selected_kinds) == 0:
            self._show_warning("Select at least one result type to extract.", "Missing Result Type")
            return None

        try:
            selected_base_output_dir = self._ensure_output_base_dir(selected_base_output_dir)
        except Exception as out_ex:
            self._show_warning(
                "Could not access selected output folder: {0}".format(str(out_ex)),
                "Invalid Output Folder",
            )
            return None

        try:
            selected_mode_limit = self.selected_mode_limit()
        except Exception as mode_ex:
            self._show_warning(str(mode_ex), "Invalid Mode Limit")
            return None

        submitted_epoch = time.time()
        record = {
            "job_id": int(self._next_job_id),
            "analysis": selected_analysis,
            "analysis_name": str(selected_analysis.Name),
            "named_selection": selected_ns,
            "named_selection_name": str(selected_ns.Name),

            "use_no_graphics": self.use_no_graphics_enabled(),
            "ram_strategy": selected_strategy_key,
            "mode_limit": selected_mode_limit,
            "mode_limit_max": int(self.mode_limit_spinner.Maximum),
            "available_mode_count": -1,
            "base_output_dir": selected_base_output_dir,
            "submitted_epoch": submitted_epoch,
            "status": JOB_STATUS_PENDING,
            "output_dir": "",
            "output_csv_paths": [],
            "error": "",
            "started_epoch": 0.0,
            "completed_epoch": 0.0,
        }
        self._next_job_id += 1

    def _on_submit_job_clicked(self, sender, args):
        if self._is_running:
            return
        new_job = self._make_job_record()
        if new_job is None:
            return

        self._queued_jobs.append(new_job)
        self._append_job_to_table(new_job)
        self._refresh_action_buttons()

        result_labels = get_result_kind_labels(new_job.get("result_kinds", []))
        base_output_dir = str(new_job.get("base_output_dir", "")).strip()
        if base_output_dir == "":
            base_output_dir = "(analysis ResultFileDirectory)"
        job_strategy = get_job_ram_strategy(new_job)
        mode_limit_text = format_mode_subset_with_available(
            new_job.get("mode_limit", None),
            new_job.get("mode_limit_max", -1),
        )

        queue_msg = "Job {0} submitted | Analysis: {1} | NS: {2} | Result types: {3} | No Graphics: {4} | Mode limit: {5} | RAM strategy: {6} | Save base: {7}".format(

            mode_limit_text,
            job_strategy.get("label", "Balanced"),

    def _on_run_jobs_clicked(self, sender, args):
        if self._is_running:
            return
        if self._pending_jobs_count() == 0:
            self._show_warning("No pending jobs found. Use 'Submit Jobs' before running.", "No Pending Jobs")
            return

        run_cancel_was_requested = False
        self.reset_cancel_request()
        self._is_running = True
        self._set_extract_controls_enabled(False)
        self.TopMost = True
        self._ensure_visible_foreground()
        start_msg = "Starting extraction jobs ({0} pending)...".format(self._pending_jobs_count())
        self.update_progress(start_msg, 0, 1000)

        try:
            self._run_callback(self)
        except Exception as ex:
            err = "Extraction jobs failed: {0}".format(str(ex))
            self.update_progress(err)
            self._show_error(err, "Modal Extractor Error")
        finally:
            run_cancel_was_requested = bool(self._cancel_requested)
            self._is_running = False
            self.TopMost = False
            self.reset_cancel_request()
            self._set_extract_controls_enabled(True)
            if run_cancel_was_requested:
                self.progress_bar.Value = 0
            Application.DoEvents()

    def _on_cancel_clicked(self, sender, args):
        if not self._is_running:
            return
        if self._cancel_requested:
            return
        self._cancel_requested = True
        self._refresh_action_buttons()
        self.update_progress("Cancellation requested... stopping at next safe checkpoint.")

        if self._is_running:
            self._show_warning("Extraction is still running. Please wait until it finishes.", "Please Wait")
            return
        self.DialogResult = DialogResult.Cancel
        self.Close()

def create_component_objects(solution, named_selection, result_kind):
    component_names = []
    component_objects = []

    with begin_transaction(True):
        if result_kind == RESULT_KIND_STRESS:
            definitions = [
                ("sx", "normal", NormalOrientationType.XAxis),
                ("sy", "normal", NormalOrientationType.YAxis),
                ("sz", "normal", NormalOrientationType.ZAxis),
                ("sxy", "shear", ShearOrientationType.XYPlane),
                ("syz", "shear", ShearOrientationType.YZPlane),
                ("sxz", "shear", ShearOrientationType.XZPlane),
            ]
            for comp_name, kind, orient in definitions:
                if kind == "normal":
                    obj = solution.AddNormalStress()
                    obj.NormalOrientation = orient
                else:
                    obj = solution.AddShearStress()
                    obj.ShearOrientation = orient
                obj.Name = "__TMP_MODAL_{0}".format(comp_name.upper())
                apply_named_selection_scope(obj, named_selection)
                component_names.append(comp_name)
                component_objects.append(obj)

        elif result_kind == RESULT_KIND_DEFORMATION:
            definitions = [
                ("ux", NormalOrientationType.XAxis),
                ("uy", NormalOrientationType.YAxis),
                ("uz", NormalOrientationType.ZAxis),
            ]
            for comp_name, orient in definitions:
                obj = solution.AddDirectionalDeformation()
                obj.NormalOrientation = orient
                obj.Name = "__TMP_MODAL_{0}".format(comp_name.upper())
                apply_named_selection_scope(obj, named_selection)
                component_names.append(comp_name)
                component_objects.append(obj)

        elif result_kind == RESULT_KIND_FORCE_MOMENT:
            global_cs = get_global_coordinate_system_object()
            if global_cs is None:
                raise Exception(
                    "Global Coordinate System was not found. Cannot extract ENFO/ENMO in global CS."
                )

            definitions = [
                ("enfox", "ENFOX"),
                ("enfoy", "ENFOY"),
                ("enfoz", "ENFOZ"),
                ("enmox", "ENMOX"),
                ("enmoy", "ENMOY"),
                ("enmoz", "ENMOZ"),
            ]
            for comp_name, expression in definitions:
                obj = solution.AddUserDefinedResult()
                obj.Expression = expression
                obj.CoordinateSystem = global_cs
                obj.Name = "__TMP_MODAL_{0}".format(comp_name.upper())
                apply_named_selection_scope(obj, named_selection)
                component_names.append(comp_name)
                component_objects.append(obj)
        else:
            raise Exception("Unknown result kind: {0}".format(result_kind))

    return component_names, component_objects


def cleanup_component_objects(component_objects):
    if component_objects is None:
        return
    try:
        graphics_ctx = begin_graphics_suspend()
    except Exception:
        graphics_ctx = _NoOpTransaction()
    try:
        tx_ctx = begin_transaction(True)
    except Exception:
        tx_ctx = _NoOpTransaction()

    with graphics_ctx:
        with tx_ctx:
            for obj in component_objects:
                try:
                    obj.Delete()
                except Exception:
                    pass
    force_memory_cleanup(True)

def normalize_result_unit_text(unit_text):
    text = str(unit_text).strip().lower()
    text = text.replace("[", "").replace("]", "")
    text = text.replace("(", "").replace(")", "")
    text = text.replace(" ", "")
    text = text.replace("²", "^2")
    text = text.replace("³", "^3")
    return text


def is_expected_stress_unit(normalized_unit_text):
    if normalized_unit_text == "mpa":
        return True
    if normalized_unit_text in ("n/mm^2", "nmm^-2", "n/mm/mm"):
        return True
    return False


def is_expected_deformation_unit(normalized_unit_text):
    return normalized_unit_text == "mm"


def get_scale_factors(result_kind, component_count, unit_text, progress_dialog=None):
    factors = [1.0] * component_count
    unit_text_display = str(unit_text).strip()
    unit_normalized = normalize_result_unit_text(unit_text)

    if result_kind == RESULT_KIND_STRESS:
        if is_expected_stress_unit(unit_normalized):
            log_message(
                "Validated stress result unit after unit-system switch: {0}.".format(
                    unit_text_display if unit_text_display != "" else EXPECTED_RESULT_UNIT_LABELS[RESULT_KIND_STRESS]
                ),
                progress_dialog,
            )
        elif unit_normalized == "pa":
            log_message(
                "Warning: stress result unit remained '{0}' after forcing {1}. "
                "Continuing with explicit Pa -> MPa conversion.".format(
                    unit_text_display if unit_text_display != "" else "Pa",
                    UNIT_SYSTEM_STANDARD_NMM_LABEL,
                ),
                progress_dialog,
            )
            factors = [1e-6] * component_count
        else:
            raise Exception(
                "Unexpected stress result unit '{0}' after forcing {1}. "
                "Expected MPa-compatible units (for example MPa or N/mm^2).".format(
                    unit_text_display if unit_text_display != "" else "(blank)",
                    UNIT_SYSTEM_STANDARD_NMM_LABEL,
                )
            )
    elif result_kind == RESULT_KIND_DEFORMATION:
        if is_expected_deformation_unit(unit_normalized):
            log_message(
                "Validated deformation result unit after unit-system switch: {0}.".format(
                    unit_text_display if unit_text_display != "" else EXPECTED_RESULT_UNIT_LABELS[RESULT_KIND_DEFORMATION]
                ),
                progress_dialog,
            )
        elif unit_normalized == "m":
            log_message(
                "Warning: deformation result unit remained '{0}' after forcing {1}. "
                "Continuing with explicit m -> mm conversion.".format(
                    unit_text_display if unit_text_display != "" else "m",
                    UNIT_SYSTEM_STANDARD_NMM_LABEL,
                ),
                progress_dialog,
            )
            factors = [1e3] * component_count
        else:
            raise Exception(
                "Unexpected deformation result unit '{0}' after forcing {1}. "
                "Expected mm.".format(
                    unit_text_display if unit_text_display != "" else "(blank)",
                    UNIT_SYSTEM_STANDARD_NMM_LABEL,
                )
            )
    return factors


def parse_float_token(token_text):
    return float(str(token_text).replace("D", "E").replace("d", "e"))


def extract_named_selection_node_ids(named_selection):
    try:
        ids = named_selection.Location.Ids
    except Exception:
        ids = None

    if ids is None or len(ids) == 0:
        raise Exception(
            "Named selection '{0}' does not provide node IDs for fallback extraction.".format(
                getattr(named_selection, "Name", "")
            )
        )

    unique = {}
    node_ids = []
    for raw_id in ids:
        try:
            nid = int(raw_id)
        except Exception:
            continue
        if nid not in unique:
            unique[nid] = True
            node_ids.append(nid)

    if len(node_ids) == 0:
        raise Exception(
            "Named selection '{0}' did not resolve to valid node IDs.".format(
                getattr(named_selection, "Name", "")
            )
        )

    node_ids.sort()
    return node_ids


def export_result_to_text_file(result_object, export_path):
    if os.path.exists(export_path):
        safe_delete_file(export_path)

    exporters = ["ExportToTextFile", "ExportToFile", "Export"]
    last_error = None

    for method_name in exporters:
        method = getattr(result_object, method_name, None)
        if method is None:
            continue

        try:
            method(export_path)
            if os.path.exists(export_path):
                return export_path
        except Exception as ex:
            last_error = ex

    if last_error is not None:
        raise Exception(
            "Failed to export result data to text file using {0}. Detail: {1}".format(
                ", ".join(exporters), str(last_error)
            )
        )

    raise Exception("Failed to export result data to text file: no export method available.")


def extract_aligned_values_from_text_export(
    result_object,
    base_node_ids,
    scale_factor,
    export_path,
    component_name="",
    progress_dialog=None,
):
    if base_node_ids is None or len(base_node_ids) == 0:
        raise Exception("Base node IDs are empty.")

    export_result_to_text_file(result_object, export_path)

    try:
        base_node_id_set = set([int(nid) for nid in base_node_ids])
        sum_map = {}
        count_map = {}
        line_index = 0

        with open(export_path, "rb") as f:
            for raw_line in f:
                line_index += 1
                if line_index % 5000 == 0:
                    raise_if_run_cancel_requested(
                        progress_dialog,
                        "while parsing text export for component '{0}'".format(component_name),
                    )
                try:
                    line_text = raw_line.decode("utf-8", "ignore")
                except Exception:
                    line_text = str(raw_line)

                tokens = NUMERIC_TOKEN_PATTERN.findall(line_text)
                if len(tokens) < 2:
                    continue

                try:
                    value = parse_float_token(tokens[-1])
                except Exception:
                    continue

                node_id = None
                for i in xrange(len(tokens) - 1):
                    token = tokens[i]
                    try:
                        node_float = parse_float_token(token)
                    except Exception:
                        continue

                    node_int = int(node_float)
                    if abs(node_float - float(node_int)) > 1e-9:
                        continue
                    if node_int in base_node_id_set:
                        node_id = node_int
                        break

                if node_id is None:
                    continue

                if node_id in sum_map:
                    sum_map[node_id] += value
                    count_map[node_id] += 1
                else:
                    sum_map[node_id] = value
                    count_map[node_id] = 1

        if len(sum_map) == 0:
            if component_name is None or str(component_name).strip() == "":
                component_name = "(unknown)"
            raise Exception(
                "Text export parsing returned no usable node-value rows for component '{0}'.".format(component_name)
            )

        aligned = []
        matched = 0
        for nid in base_node_ids:
            nid_int = int(nid)
            if nid_int in sum_map:
                matched += 1
                aligned.append((sum_map[nid_int] / float(max(1, count_map[nid_int]))) * scale_factor)
            else:
                aligned.append(0.0)

        if matched == 0:
            raise Exception("Text export parsing found no rows matching the base node scope.")

        return aligned
    finally:
        safe_delete_file(export_path)


def extract_node_ids_from_result(result_object):
    independent = result_object.PlotData.Independents
    candidate_keys = ["Node", "Nodes", "NodeID", "Node ID"]
    available_keys = [str(k) for k in independent.Keys]

    for key in candidate_keys:
        if key in available_keys:
            arr = independent[key]
            return [int(x) for x in arr]

    if len(available_keys) > 0:
        arr = independent[available_keys[0]]
        return [int(x) for x in arr]

    raise Exception("Could not extract node IDs from PlotData.Independents.")


def extract_scaled_values(result_object, scale_factor):
    values_raw = result_object.PlotData.Dependents["Values"]
    scaled = [float(v.Value if hasattr(v, "Value") else v) * scale_factor for v in values_raw]
    values_raw = None
    return scaled


def extract_aligned_values(result_object, base_node_ids, scale_factor, return_order_match=False):
    values = extract_scaled_values(result_object, scale_factor)
    current_node_ids = extract_node_ids_from_result(result_object)

    if len(current_node_ids) == len(base_node_ids):
        is_same_order = True
        if len(current_node_ids) > 0:
            if current_node_ids[0] != base_node_ids[0] or current_node_ids[-1] != base_node_ids[-1]:
                is_same_order = False
        if is_same_order:
            if return_order_match:
                return values, True
            return values

    value_map = {}
    n = min(len(current_node_ids), len(values))
    for i in xrange(n):
        value_map[int(current_node_ids[i])] = values[i]
    aligned_values = [value_map.get(int(nid), 0.0) for nid in base_node_ids]

    if return_order_match:
        return aligned_values, False
    return aligned_values


def get_node_coordinates(node_ids, progress_dialog=None):
    raise_if_run_cancel_requested(progress_dialog, "before coordinate extraction")
    mesh_data = get_mesh_data_global()
    coords = []

    if mesh_data is None:
        log_message("MeshDataByName('Global') unavailable. Coordinate columns will be blank.", progress_dialog)
        for _ in node_ids:
            coords.append(("", "", ""))
        return coords

    total = len(node_ids)
    for i in xrange(total):
        if i > 0 and i % 2000 == 0:
            raise_if_run_cancel_requested(progress_dialog, "during coordinate extraction")
        nid = int(node_ids[i])
        node = mesh_data.NodeById(nid)
        if node is None:
            x = ""
            y = ""
            z = ""
        else:
            x = float(node.X.Value if hasattr(node.X, "Value") else node.X)
            y = float(node.Y.Value if hasattr(node.Y, "Value") else node.Y)
            z = float(node.Z.Value if hasattr(node.Z, "Value") else node.Z)
        coords.append((x, y, z))

        if i > 0 and i % 20000 == 0:
            msg = "Coordinates: {0}/{1} nodes".format(i, total)
            log_message(msg, progress_dialog)
            if progress_dialog is not None:
                progress_dialog.update_progress(msg)

    return coords

            writer.writerow([node_ids[i], c[0], c[1], c[2]])


def write_batch_csv(path, batch_modes, component_names, node_ids, batch_data):
    with open(path, "wb") as f:
        writer = csv.writer(f)

        header = ["NodeID"]
        for mode in batch_modes:
            for comp_name in component_names:
                header.append("{0}_Mode{1}".format(comp_name, mode))
        writer.writerow(header)

        node_count = len(node_ids)
        comp_count = len(component_names)
        for row_index in xrange(node_count):
            row = [node_ids[row_index]]
            for mode in batch_modes:
                mode_components = batch_data[mode]
                for comp_idx in xrange(comp_count):
                    row.append(mode_components[comp_idx][row_index])
            writer.writerow(row)

def should_use_solution_evaluate(solution, temp_component_count):
    children_count = len(solution.Children)
    # If there are many existing results, evaluating the full solution can be slower.
    return children_count <= (temp_component_count + 4)


def merge_pass(base_csv_path, batch_csv_paths, output_csv_path, node_count, pass_index, total_passes, progress_dialog):
    base_handle = None
    out_handle = None
    batch_handles = []
    batch_readers = []
    warned_node_mismatch = False

    try:
        base_handle = open(base_csv_path, "rb")
        out_handle = open(output_csv_path, "wb")
        base_reader = csv.reader(base_handle)
        writer = csv.writer(out_handle)

        base_header = base_reader.next()
        merged_header = list(base_header)

        for batch_path in batch_csv_paths:
            h = open(batch_path, "rb")
            batch_handles.append(h)
            r = csv.reader(h)
            batch_readers.append(r)
            h_row = r.next()
            if len(h_row) > 1:
                merged_header.extend(h_row[1:])

        writer.writerow(merged_header)

        progress_interval = max(1, int(node_count / 20))
        raise_if_run_cancel_requested(progress_dialog, "before merge pass {0}".format(pass_index))
        for i in xrange(node_count):
            if i > 0 and i % progress_interval == 0:
                raise_if_run_cancel_requested(
                    progress_dialog,
                    "during merge pass {0}".format(pass_index),
                )
            base_row = base_reader.next()
            base_node = base_row[0] if len(base_row) > 0 else ""
            out_row = list(base_row)

            for r in batch_readers:
                mode_row = r.next()
                if len(mode_row) > 0 and mode_row[0] != base_node and not warned_node_mismatch:
                    warned_node_mismatch = True
                    log_message(
                        "Warning: Node ID ordering mismatch detected while merging temporary files.",
                        progress_dialog,
                    )
                if len(mode_row) > 1:
                    out_row.extend(mode_row[1:])
            writer.writerow(out_row)

            if i > 0 and i % progress_interval == 0:
                pct = int((float(i) / float(node_count)) * 100.0)
                msg = "Merging pass {0}/{1}: {2}%".format(pass_index, total_passes, pct)
                if progress_dialog is not None:
                    progress_dialog.update_progress(msg)

    finally:
        if base_handle is not None:
            base_handle.close()
        if out_handle is not None:
            out_handle.close()
        for h in batch_handles:
            try:
                h.close()
            except Exception:
                pass


def merge_component_files_into_mode_csv(component_csv_paths, output_csv_path, node_count):
    if component_csv_paths is None or len(component_csv_paths) == 0:
        raise Exception("No component CSV files were created for mode merge.")

    if os.path.exists(output_csv_path):
        os.remove(output_csv_path)

    if len(component_csv_paths) == 1:
        os.rename(component_csv_paths[0], output_csv_path)
        return output_csv_path

    try:
        merge_pass(
            component_csv_paths[0],
            component_csv_paths[1:],
            output_csv_path,
            node_count,
            1,
            1,
            None,
        )
        return output_csv_path
    finally:
        for component_path in component_csv_paths:
            if component_path != output_csv_path:
                safe_delete_file(component_path)


def safe_delete_file(path):
    try:
        if os.path.exists(path):
            os.remove(path)
    except Exception:
        pass


def cleanup_temp_folder(temp_dir):
    if temp_dir is None:
        return
    try:
        if os.path.isdir(temp_dir):
            for name in os.listdir(temp_dir):
                safe_delete_file(os.path.join(temp_dir, name))
            os.rmdir(temp_dir)
    except Exception:
        pass


def merge_batches_into_output(
    temp_dir,
    base_csv_path,
    batch_files,
    output_csv_path,
    node_count,
    progress_dialog=None,
    low_ram_free_percent=None,
):
    if len(batch_files) == 0:
        if os.path.exists(output_csv_path):
            os.remove(output_csv_path)
        os.rename(base_csv_path, output_csv_path)
        return output_csv_path

    current_base = base_csv_path
    pending = list(batch_files)
    pass_index = 0
    previous_group_limit = -1

    while len(pending) > 0:
        raise_if_run_cancel_requested(progress_dialog, "before merge pass scheduling")
        group_limit, merge_available_ram = estimate_merge_group_limit(
            len(pending),
            low_ram_free_percent=low_ram_free_percent,
        )
        if group_limit != previous_group_limit:
            log_message(
                "Merge chunk size selected: {0} files/pass (available RAM: {1}).".format(
                    group_limit,
                    format_available_ram_text(merge_available_ram),
                ),
                progress_dialog,
            )
            previous_group_limit = group_limit

        apply_low_ram_backpressure(
            progress_dialog,
            "before merge pass {0}".format(pass_index + 1),
            low_ram_free_percent=low_ram_free_percent,
        )

        pass_index += 1
        group = pending[:group_limit]
        pending = pending[group_limit:]
        remaining_passes = int(math.ceil(float(len(pending)) / float(max(group_limit, 1))))
        total_passes = pass_index + remaining_passes

        next_base = os.path.join(temp_dir, "merge_pass_{0:03d}.csv".format(pass_index))
        msg = "Merging temporary CSVs (pass {0}/{1}, files in pass: {2})".format(
            pass_index, total_passes, len(group)
        )
        log_message(msg, progress_dialog)
        if progress_dialog is not None:
            progress_dialog.update_progress(msg)

        merge_pass(
            current_base,
            group,
            next_base,
            node_count,
            pass_index,
            total_passes,
            progress_dialog,
        )

        if current_base != base_csv_path:
            safe_delete_file(current_base)
        else:
            safe_delete_file(base_csv_path)
        for fp in group:
            safe_delete_file(fp)
        current_base = next_base
        force_memory_cleanup()

    if os.path.exists(output_csv_path):
        os.remove(output_csv_path)
    os.rename(current_base, output_csv_path)
    return output_csv_path

def collect_job_container_folders(job_records):
    folders = []
    seen = {}
    if job_records is None:
        return folders

    for job in job_records:
        if job is None:
            continue

        output_dir = str(job.get("output_dir", "")).strip()
        if output_dir == "":
            continue

        try:
            folder = os.path.abspath(os.path.dirname(output_dir))
        except Exception:
            folder = os.path.dirname(output_dir)

        if folder == "" or not os.path.isdir(folder):
            continue

        key = folder.lower()
        if key in seen:
            continue

        seen[key] = True
        folders.append(folder)

    return folders


def open_folder_in_explorer(folder_path, progress_dialog=None):
    folder = str(folder_path).strip()
    if folder == "":
        return False
    if not os.path.isdir(folder):
        return False

    try:
        os.startfile(folder)
        return True
    except Exception:
        pass

    try:
        clr.AddReference("System")
        from System.Diagnostics import Process
        Process.Start(folder)
        return True
    except Exception as ex:
        log_message(
            "Could not open folder in file explorer: {0}. Detail: {1}".format(folder, str(ex)),
            progress_dialog,
        )
        return False


def open_job_container_folders(job_records, progress_dialog=None):
    folders = collect_job_container_folders(job_records)
    opened_folders = []
    for folder in folders:
        if open_folder_in_explorer(folder, progress_dialog):
            opened_folders.append(folder)
            log_message("Opened job container folder: {0}".format(folder), progress_dialog)
    return opened_folders

def run_extraction(
    solution,
    named_selection,
    result_kind,
    mode_ids,
    progress_dialog=None,
    result_kind_label=None,
    output_dir=None,
    output_name_suffix="",
    ram_strategy=None,
):
    start_time = time.time()
    temp_objects = []
    temp_dir = None

    component_names = []
    component_objects = []
    batch_files = []
    node_ids = []
    base_csv_path = None
    use_plotdata_values = True
    plotdata_fallback_logged = False
    if result_kind_label is None:
        result_kind_label = RESULT_KIND_LABEL_BY_ID.get(result_kind, result_kind)

    if ram_strategy is None:
        ram_strategy = get_ram_strategy(RAM_STRATEGY_DEFAULT)
    effective_gc_interval = int(ram_strategy.get("gc_interval", GC_INTERVAL_MODES))
    total_ram_bytes = get_total_ram_bytes()

    # Dynamic progress allocation: extraction ~77%, merge ~23%.
    total_modes = len(mode_ids)
    extraction_ticks = int(1000.0 * (1.0 / 1.3))
    merge_ticks = 1000 - extraction_ticks

    try:
        raise_if_run_cancel_requested(progress_dialog, "before extraction setup")
        msg = "Creating temporary result objects..."
        log_message(msg, progress_dialog)
        if progress_dialog is not None:
            progress_dialog.update_progress(msg, 0, 1000)

        strategy_label = ram_strategy.get("label", "Balanced")
        trigger_pct = ram_strategy.get("low_ram_free_percent", 35.0)
        if total_ram_bytes > 0:
            log_message(
                "RAM strategy: {0} (low-RAM trigger: <= {1:.1f}% free, GC interval: {2} modes).".format(
                    strategy_label, trigger_pct, effective_gc_interval
                ),
                progress_dialog,
            )
        else:
            log_message(
                "RAM strategy: {0} (low-RAM trigger: <= {1:.1f}% free, GC interval: {2} modes; "
                "total RAM query unavailable; fallback trigger below {3} free).".format(
                    strategy_label,
                    trigger_pct,
                    effective_gc_interval,
                    format_available_ram_text(LOW_RAM_BACKPRESSURE_FALLBACK_BYTES),
                ),
                progress_dialog,
            )

        with begin_graphics_suspend():
            component_names, component_objects = create_component_objects(solution, named_selection, result_kind)
            temp_objects = component_objects
            comp_count = len(component_objects)
            if comp_count == 0:
                raise Exception("No temporary extraction result objects were created.")

            first_mode = int(mode_ids[0])
            with begin_transaction(True):
                component_objects[0].SetNumber = first_mode
            component_objects[0].EvaluateAllResults()

            try:
                node_ids = extract_node_ids_from_result(component_objects[0])
            except Exception as node_ex:
                if result_kind == RESULT_KIND_FORCE_MOMENT:
                    node_ids = extract_named_selection_node_ids(named_selection)
                    use_plotdata_values = False
                    log_message(
                        "PlotData is unavailable for ENFO/ENMO in this environment. "
                        "Switching to ExportToTextFile fallback. Detail: {0}".format(str(node_ex)),
                        progress_dialog,
                    )
                    plotdata_fallback_logged = True
                else:
                    raise

            if len(node_ids) == 0:
                raise Exception("No nodes found in extracted scope.")
            first_unit = get_values_unit(component_objects[0])
            clear_component_generated_data(component_objects)
            force_memory_cleanup()

            scale_factors = get_scale_factors(
                result_kind,
                comp_count,
                first_unit,
                progress_dialog=progress_dialog,
            )

            msg = "Extracting coordinates for {0} scoped nodes...".format(len(node_ids))
            log_message(msg, progress_dialog)
            if progress_dialog is not None:
                progress_dialog.update_progress(msg, 5, 1000)
            node_coords = get_node_coordinates(node_ids, progress_dialog)
            raise_if_run_cancel_requested(progress_dialog, "after coordinate extraction")

            temp_folder_name = "_tensor_extract_tmp_{0}".format(int(time.time()))
            temp_parent_dir = get_solution_default_io_directory(solution)
            temp_dir = os.path.join(temp_parent_dir, temp_folder_name)
            try:
                if not os.path.isdir(temp_dir):
                    os.makedirs(temp_dir)
            except Exception as temp_ex:
                fallback_temp_parent = get_solution_working_directory(solution)
                if fallback_temp_parent != "" and fallback_temp_parent != temp_parent_dir:
                    log_message(
                        "Failed creating temp folder under ResultFileDirectory. "
                        "Falling back to WorkingDir. Detail: {0}".format(str(temp_ex)),
                        progress_dialog,
                    )
                    temp_dir = os.path.join(fallback_temp_parent, temp_folder_name)
                    if not os.path.isdir(temp_dir):
                        os.makedirs(temp_dir)
                else:
                    raise

            base_csv_path = os.path.join(temp_dir, "base_coords.csv")
            write_base_coordinate_csv(base_csv_path, node_ids, node_coords)
            del node_coords
            force_memory_cleanup()

            use_solution_eval = False
            if result_kind == RESULT_KIND_FORCE_MOMENT:
                use_solution_eval = True
                log_message(
                    "ENFO/ENMO extraction uses solution.EvaluateAllResults() per mode; failure is treated as unavailable data for that mode/scope.",
                    progress_dialog,
                )
            elif result_kind == RESULT_KIND_STRESS and ram_strategy.get("evaluate_stress_once_per_mode", False):
                use_solution_eval = True
                log_message(
                    "Stress extraction uses solution.EvaluateAllResults() once per mode for the {0} RAM strategy.".format(
                        strategy_label
                    ),
                    progress_dialog,
                )
            else:
                log_message(
                    "Using per-object EvaluateAllResults() and per-component disk spooling to minimize retained extraction data.",
                    progress_dialog,
                )
            if use_solution_eval:
                log_message("Using solution.EvaluateAllResults() once per mode.", progress_dialog)

            mode_cursor = 0
            batch_index = 0
            consecutive_backpressure_count = 0
            while mode_cursor < total_modes:
                raise_if_run_cancel_requested(progress_dialog, "before mode batch setup")
                remaining = total_modes - mode_cursor

                # Backpressure with teeth: consecutive triggers force batch_size=1.
                if consecutive_backpressure_count >= 2:
                    batch_size = 1
                    available_ram = get_available_ram_bytes()
                    is_low_ram = True
                    log_message(
                        "Consecutive backpressure detected ({0}x). Forcing batch_size=1.".format(
                            consecutive_backpressure_count
                        ),
                        progress_dialog,
                    )
                else:
                    batch_size, available_ram, is_low_ram = estimate_mode_batch_size(
                        len(node_ids),
                        comp_count,
                        remaining,
                        ram_strategy=ram_strategy,
                    )
                if batch_size > remaining:
                    batch_size = remaining

                batch_modes = mode_ids[mode_cursor: mode_cursor + batch_size]
                batch_index += 1

                # Determine GC interval for this batch based on RAM state.
                batch_gc_interval = GC_INTERVAL_LOW_RAM_MODES if is_low_ram else effective_gc_interval

                if available_ram > 0:
                    ram_text = "{0:.2f} GiB".format(bytes_to_gib(available_ram))
                else:
                    ram_text = "unknown"
                msg = "Batch {0}: modes {1}-{2} ({3} modes), available RAM: {4}".format(
                    batch_index,
                    batch_modes[0],
                    batch_modes[-1],
                    len(batch_modes),
                    ram_text,
                )
                log_message(msg, progress_dialog)

                try:
                    for local_idx in xrange(len(batch_modes)):
                        mode = int(batch_modes[local_idx])
                        global_mode_idx = mode_cursor + local_idx + 1
                        raise_if_run_cancel_requested(progress_dialog, "before extracting mode {0}".format(mode))
                        mode_msg = "Extracting [{0}] mode {1}/{2} (set {3})".format(
                            result_kind_label,
                            global_mode_idx,
                            total_modes,
                            mode,
                        )
                        log_message(mode_msg, progress_dialog)
                        if progress_dialog is not None:
                            mode_progress = int((float(global_mode_idx) / float(total_modes)) * float(extraction_ticks))
                            progress_dialog.update_progress(mode_msg, mode_progress, 1000)

                        with begin_transaction(True):
                            for obj in component_objects:
                                obj.SetNumber = mode

                        if use_solution_eval:
                            try:
                                solution.EvaluateAllResults()
                            except Exception as eval_all_ex:
                                if result_kind == RESULT_KIND_FORCE_MOMENT:
                                    raise Exception(
                                        "ENFO/ENMO evaluation failed at mode {0}. "
                                        "Requested scope appears unavailable in RST for force/moment channels. Detail: {1}".format(
                                            mode, str(eval_all_ex)
                                        )
                                    )
                                else:
                                    raise
                        component_csv_paths = []
                        first_is_base_order = False
                        try:
                            for comp_idx in xrange(comp_count):
                                obj = component_objects[comp_idx]
                                component_name = component_names[comp_idx]

                                if not use_solution_eval:
                                    obj.EvaluateAllResults()

                                if comp_idx == 0:
                                    if use_plotdata_values:
                                        try:
                                            component_values, first_is_base_order = extract_aligned_values(
                                                obj,
                                                node_ids,
                                                scale_factors[comp_idx],
                                                return_order_match=True,
                                            )
                                        except Exception as value_ex:
                                            if result_kind != RESULT_KIND_FORCE_MOMENT:
                                                raise

                                            use_plotdata_values = False
                                            if not plotdata_fallback_logged:
                                                log_message(
                                                    "PlotData access failed for ENFO/ENMO. "
                                                    "Switching to ExportToTextFile fallback for remaining extraction. "
                                                    "Detail: {0}".format(str(value_ex)),
                                                    progress_dialog,
                                                )
                                                plotdata_fallback_logged = True

                                            export_path = os.path.join(
                                                temp_dir,
                                                "__tmp_export_mode_{0:05d}_{1}.txt".format(mode, component_name),
                                            )
                                            component_values = extract_aligned_values_from_text_export(
                                                obj,
                                                node_ids,
                                                scale_factors[comp_idx],
                                                export_path,
                                                component_name=component_name,
                                                progress_dialog=progress_dialog,
                                            )
                                            first_is_base_order = False
                                    else:
                                        export_path = os.path.join(
                                            temp_dir,
                                            "__tmp_export_mode_{0:05d}_{1}.txt".format(mode, component_name),
                                        )
                                        component_values = extract_aligned_values_from_text_export(
                                            obj,
                                            node_ids,
                                            scale_factors[comp_idx],
                                            export_path,
                                            component_name=component_name,
                                            progress_dialog=progress_dialog,
                                        )
                                        first_is_base_order = False
                                else:
                                    if use_plotdata_values:
                                        if first_is_base_order:
                                            component_values = extract_scaled_values(obj, scale_factors[comp_idx])
                                        else:
                                            component_values = extract_aligned_values(obj, node_ids, scale_factors[comp_idx])
                                    else:
                                        export_path = os.path.join(
                                            temp_dir,
                                            "__tmp_export_mode_{0:05d}_{1}.txt".format(mode, component_name),
                                        )
                                        component_values = extract_aligned_values_from_text_export(
                                            obj,
                                            node_ids,
                                            scale_factors[comp_idx],
                                            export_path,
                                            component_name=component_name,
                                            progress_dialog=progress_dialog,
                                        )

                                component_csv_path = os.path.join(
                                    temp_dir,
                                    "__tmp_mode_{0:05d}_{1}.csv".format(mode, component_name),
                                )
                                write_component_csv(
                                    component_csv_path,
                                    mode,
                                    component_name,
                                    node_ids,
                                    component_values,
                                )
                                component_csv_paths.append(component_csv_path)

                                component_values = None
                                try:
                                    obj.ClearGeneratedData()
                                except Exception:
                                    pass

                                raise_if_run_cancel_requested(
                                    progress_dialog,
                                    "after extracting component {0} for mode {1}".format(component_name, mode),
                                )

                            mode_file = os.path.join(temp_dir, "mode_{0:05d}.csv".format(mode))
                            merge_component_files_into_mode_csv(
                                component_csv_paths,
                                mode_file,
                                len(node_ids),
                            )
                        finally:
                            for component_csv_path in component_csv_paths:
                                safe_delete_file(component_csv_path)

                        batch_files.append(mode_file)

                        if use_solution_eval and (
                            SOLUTION_CHILDREN_CLEAR_INTERVAL <= 1
                            or (global_mode_idx % SOLUTION_CHILDREN_CLEAR_INTERVAL) == 0
                        ):
                            cleanup_solution_generated_data(solution)

                        # Interval-based GC: only collect every N modes within a batch.
                        if batch_gc_interval <= 1 or (local_idx + 1) % batch_gc_interval == 0:
                            force_memory_cleanup(False)

                        raise_if_run_cancel_requested(progress_dialog, "after extracting mode {0}".format(mode))
                finally:
                    clear_component_generated_data(component_objects)
                    force_memory_cleanup(True)

                mode_cursor += len(batch_modes)

                # Batch-boundary backpressure check with re-evaluation for next batch.
                bp_state = apply_low_ram_backpressure(
                    progress_dialog,
                    "after batch {0}".format(batch_index),
                    ram_strategy=ram_strategy,
                )
                if bp_state.get("triggered", False):
                    consecutive_backpressure_count += 1
                else:
                    consecutive_backpressure_count = 0

        output_name = RESULT_OUTPUT_FILES[result_kind]
        if output_name_suffix is not None and str(output_name_suffix).strip() != "":
            output_name = append_suffix_before_extension(output_name, str(output_name_suffix))

        if output_dir is None or str(output_dir).strip() == "":
            output_dir = get_solution_default_io_directory(solution)
        if not os.path.isdir(output_dir):
            try:
                os.makedirs(output_dir)
            except Exception as output_ex:
                fallback_output_dir = get_solution_working_directory(solution)
                if fallback_output_dir != "" and fallback_output_dir != output_dir:
                    log_message(
                        "Failed creating output folder under ResultFileDirectory. "
                        "Falling back to WorkingDir. Detail: {0}".format(str(output_ex)),
                        progress_dialog,
                    )
                    output_dir = fallback_output_dir
                    if not os.path.isdir(output_dir):
                        os.makedirs(output_dir)
                else:
                    raise
        output_csv_path = os.path.join(output_dir, output_name)

        msg = "Merging temporary CSV chunks into final output..."
        log_message(msg, progress_dialog)
        if progress_dialog is not None:
            progress_dialog.update_progress(msg, extraction_ticks, 1000)
        raise_if_run_cancel_requested(progress_dialog, "before merge phase")

        merge_batches_into_output(
            temp_dir=temp_dir,
            base_csv_path=base_csv_path,
            batch_files=batch_files,
            output_csv_path=output_csv_path,
            node_count=len(node_ids),
            progress_dialog=progress_dialog,
            low_ram_free_percent=ram_strategy.get("low_ram_free_percent", 35.0),
        )

        if progress_dialog is not None:
            progress_dialog.update_progress("Completed.", 1000, 1000)

        elapsed = time.time() - start_time
        log_message("Extraction completed in {0:.1f} s".format(elapsed), progress_dialog)
        log_message("CSV file saved at: {0}".format(output_csv_path), progress_dialog)
        return output_csv_path

    finally:
        cleanup_component_objects(temp_objects)
        cleanup_temp_folder(temp_dir)
        restore_ui_interaction_state()
        force_memory_cleanup(True)


def main():
    modal_analyses = get_modal_analysis_candidates()
    if len(modal_analyses) == 0:
        raise Exception("No modal analysis environment found in Mechanical tree.")

    nodal_named_selections = get_nodal_named_selections()
    if len(nodal_named_selections) == 0:
        raise Exception("No named selections found in tree.")

    def run_from_dialog(dialog):
        pending_jobs = dialog.pending_jobs()
        if len(pending_jobs) == 0:
            raise Exception("No pending jobs found. Use 'Submit Jobs' before running.")

        total_jobs = len(pending_jobs)
        log_message("Starting extraction queue with {0} pending jobs.".format(total_jobs), dialog)
        log_message(
            "Jobs are processed sequentially with per-kind and per-job cleanup to keep RAM usage stable.",
            dialog,
        )

        completed_job_ids = []
        failed_jobs = []
        cancelled_jobs = []
        all_output_csvs = []
        cancellation_message = ""

        for job_idx in xrange(total_jobs):
            if dialog.is_cancel_requested():
                cancellation_message = "Cancellation requested by user. Remaining pending jobs were not started."
                log_message(cancellation_message, dialog)
                break

            job = pending_jobs[job_idx]
            job_id = int(job.get("job_id", job_idx + 1))
            selected_analysis = job.get("analysis", None)
            selected_ns = job.get("named_selection", None)
            selected_kinds = list(job.get("result_kinds", []))
            use_no_graphics_pref = bool(job.get("use_no_graphics", False))
            job_ram_strategy = get_job_ram_strategy(job)
            mode_limit_requested = job.get("mode_limit", None)
            mode_limit_text = ""

            if selected_analysis is None or selected_ns is None or len(selected_kinds) == 0:
                dialog.set_job_status(job_id, JOB_STATUS_FAILED)
                cfg_err = "Job configuration is incomplete."
                job["error"] = cfg_err
                failed_jobs.append((job_id, cfg_err))
                log_message("Job {0} failed before start: {1}".format(job_id, cfg_err), dialog)
                continue

            pref_state = None
            unit_state = None
            solution = None
            mode_ids = []
            output_dir = None
            output_csvs = []
            job_start = time.time()
            job_end = job_start
            available_ram_start = get_available_ram_bytes()

            job["started_epoch"] = job_start
            job["completed_epoch"] = 0.0
            job["error"] = ""
            dialog.set_job_status(job_id, JOB_STATUS_RUNNING)

            try:
                dialog.raise_if_cancel_requested("before starting job {0}".format(job_id))
                solution, mode_ids = get_analysis_solution_and_modes(selected_analysis)
                all_mode_ids = list(mode_ids)
                available_mode_count = len(all_mode_ids)
                job["available_mode_count"] = available_mode_count
                mode_ids = all_mode_ids

                if mode_limit_requested is not None:
                    try:
                        requested_n_modes = int(mode_limit_requested)
                    except Exception:
                        raise Exception(
                            "Job {0} has invalid mode limit setting: {1}".format(job_id, str(mode_limit_requested))
                        )
                    if requested_n_modes <= 0:
                        raise Exception("Job {0} has non-positive mode limit setting.".format(job_id))
                    if requested_n_modes < available_mode_count:
                        mode_ids = all_mode_ids[:requested_n_modes]
                    elif requested_n_modes > available_mode_count:
                        log_message(
                            "Job {0}: requested first {1} modes, but only {2} are available. Extracting first {2} modes.".format(
                                job_id, requested_n_modes, available_mode_count
                            ),
                            dialog,
                        )
                mode_limit_text = format_mode_subset_with_available(mode_limit_requested, available_mode_count)

                selected_base_output_dir = str(job.get("base_output_dir", "")).strip()
                if selected_base_output_dir == "":
                    selected_base_output_dir = get_solution_default_io_directory(solution)
                if not os.path.isdir(selected_base_output_dir):
                    try:
                        os.makedirs(selected_base_output_dir)
                    except Exception as output_base_ex:
                        fallback_output_base = get_solution_working_directory(solution)
                        if fallback_output_base != "" and fallback_output_base != selected_base_output_dir:
                            log_message(
                                "Failed creating base output folder under ResultFileDirectory. "
                                "Falling back to WorkingDir. Detail: {0}".format(str(output_base_ex)),
                                dialog,
                            )
                            selected_base_output_dir = fallback_output_base
                            if not os.path.isdir(selected_base_output_dir):
                                os.makedirs(selected_base_output_dir)
                        else:
                            raise

                job["base_output_dir"] = selected_base_output_dir
                output_dir = create_job_output_folder(selected_base_output_dir, job_id)
                job["output_dir"] = output_dir

                kind_labels = get_result_kind_labels(selected_kinds)
                start_msg = (
                    "Job {0}/{1} (ID {2}) | Analysis: {3} | NS: {4} | Result types: {5} | Modes: 1..{6} ({7}) | RAM strategy: {8}"
                ).format(
                    job_idx + 1,
                    total_jobs,
                    job_id,
                    selected_analysis.Name,
                    selected_ns.Name,
                    "; ".join(kind_labels),
                    len(mode_ids),
                    mode_limit_text,
                    job_ram_strategy.get("label", "Balanced"),
                )
                log_message(start_msg, dialog)
                log_message("Job {0} base output folder: {1}".format(job_id, selected_base_output_dir), dialog)
                log_message("Job {0} output folder: {1}".format(job_id, output_dir), dialog)
                dialog.update_progress(start_msg, 0, 1000)

                if use_no_graphics_pref:
                    pref_state = enter_no_graphics_preference_mode(dialog)
                else:
                    log_message("Job {0}: No-Graphics preference is disabled.".format(job_id), dialog)

                unit_state = enter_standard_nmm_unit_system(dialog)

                n_selected = len(selected_kinds)
                for run_idx in xrange(n_selected):
                    dialog.raise_if_cancel_requested(
                        "before extraction {0}/{1} of job {2}".format(run_idx + 1, n_selected, job_id)
                    )
                    selected_kind = selected_kinds[run_idx]
                    kind_label = RESULT_KIND_LABEL_BY_ID.get(selected_kind, selected_kind)
                    available_ram = get_available_ram_bytes()
                    stage_msg = (
                        "Job {0}/{1} (ID {2}) | Extraction {3}/{4}: {5} | Available RAM before start: {6}"
                    ).format(
                        job_idx + 1,
                        total_jobs,
                        job_id,
                        run_idx + 1,
                        n_selected,
                        kind_label,
                        format_available_ram_text(available_ram),
                    )
                    log_message(stage_msg, dialog)
                    dialog.update_progress(stage_msg, 0, 1000)

                    output_csv = run_extraction(
                        solution,
                        selected_ns,
                        selected_kind,
                        mode_ids,
                        dialog,
                        kind_label,
                        output_dir=output_dir,
                        output_name_suffix="_job_{0}".format(job_id),
                        ram_strategy=job_ram_strategy,
                    )
                    output_csvs.append(output_csv)
                    all_output_csvs.append(output_csv)

                    # Explicit cleanup between result kinds for stable memory behavior.
                    force_memory_cleanup(True)
                    time.sleep(0.1)

                dialog.raise_if_cancel_requested("after extraction loop of job {0}".format(job_id))

                job_end = time.time()
                job["completed_epoch"] = job_end
                job["output_csv_paths"] = list(output_csvs)
                dialog.set_job_status(job_id, JOB_STATUS_COMPLETED)
                completed_job_ids.append(job_id)

                write_job_metadata_file(
                    job_output_dir=output_dir,
                    job_record=job,
                    mode_count=len(mode_ids),
                    started_epoch=job_start,
                    completed_epoch=job_end,
                    status_text=JOB_STATUS_COMPLETED,
                    output_csv_paths=output_csvs,
                    available_ram_start=available_ram_start,
                )

                elapsed = job_end - job_start
                log_message("Job {0} completed in {1:.1f} s".format(job_id, elapsed), dialog)

            except UserCancelledException as cancel_ex:
                job_end = time.time()
                cancel_text = str(cancel_ex)
                if cancel_text is None or str(cancel_text).strip() == "":
                    cancel_text = "Extraction cancelled by user request."
                job["completed_epoch"] = job_end
                job["output_csv_paths"] = list(output_csvs)
                job["error"] = cancel_text
                dialog.set_job_status(job_id, JOB_STATUS_CANCELLED)
                cancelled_jobs.append((job_id, cancel_text))
                cancellation_message = cancel_text
                log_message("Job {0} cancelled: {1}".format(job_id, cancel_text), dialog)

                if output_dir is not None:
                    write_job_metadata_file(
                        job_output_dir=output_dir,
                        job_record=job,
                        mode_count=len(mode_ids),
                        started_epoch=job_start,
                        completed_epoch=job_end,
                        status_text=JOB_STATUS_CANCELLED,
                        output_csv_paths=output_csvs,
                        available_ram_start=available_ram_start,
                        error_text=cancel_text,
                    )
                break

            except Exception as job_ex:
                job_end = time.time()
                job_err = str(job_ex)
                job["completed_epoch"] = job_end
                job["output_csv_paths"] = list(output_csvs)
                job["error"] = job_err
                dialog.set_job_status(job_id, JOB_STATUS_FAILED)
                failed_jobs.append((job_id, job_err))
                log_message("Job {0} failed: {1}".format(job_id, job_err), dialog)

                if output_dir is not None:
                    write_job_metadata_file(
                        job_output_dir=output_dir,
                        job_record=job,
                        mode_count=len(mode_ids),
                        started_epoch=job_start,
                        completed_epoch=job_end,
                        status_text=JOB_STATUS_FAILED,
                        output_csv_paths=output_csvs,
                        available_ram_start=available_ram_start,
                        error_text=job_err,
                    )
            finally:
                try:
                    restore_unit_system_state(unit_state, dialog)
                except Exception:
                    pass

                try:
                    restore_no_graphics_preference_mode(pref_state, dialog)
                except Exception:
                    pass

                # Release references and force collection between jobs.
                solution = None
                mode_ids = []
                force_memory_cleanup(True)
                time.sleep(0.1)
                restore_ui_interaction_state()

        summary_lines = [
            "Extraction jobs completed.",
            "Completed jobs: {0}".format(len(completed_job_ids)),
            "Cancelled jobs: {0}".format(len(cancelled_jobs)),
            "Failed jobs: {0}".format(len(failed_jobs)),
        ]

        if cancellation_message != "":
            summary_lines.append("")
            summary_lines.append("Cancellation:")
            summary_lines.append(cancellation_message)

        if len(all_output_csvs) > 0:
            summary_lines.append("")
            summary_lines.append("Output CSV files:")
            for out_idx in xrange(len(all_output_csvs)):
                out_path = all_output_csvs[out_idx]
                summary_lines.append("{0}. {1}".format(out_idx + 1, out_path))
                summary_lines.append("")

        if len(failed_jobs) > 0:
            summary_lines.append("")
            summary_lines.append("Failed job details:")
            for failed_job_id, failed_err in failed_jobs:
                summary_lines.append("Job {0}: {1}".format(failed_job_id, failed_err))

        if len(cancelled_jobs) > 0:
            summary_lines.append("")
            summary_lines.append("Cancelled job details:")
            for cancelled_job_id, cancelled_err in cancelled_jobs:
                summary_lines.append("Job {0}: {1}".format(cancelled_job_id, cancelled_err))

        MessageBox.Show(
            dialog,
            "\n".join(summary_lines),
            "Modal Extractor",
            MessageBoxButtons.OK,
            MessageBoxIcon.Information if (len(failed_jobs) == 0 and len(cancelled_jobs) == 0) else MessageBoxIcon.Warning,
        )
        open_job_container_folders(pending_jobs, dialog)

    dlg = ExtractOptionsDialog(modal_analyses, nodal_named_selections, run_from_dialog)
    dlg.ShowDialog()
    log_message("Modal extractor dialog closed by user.")


main()
